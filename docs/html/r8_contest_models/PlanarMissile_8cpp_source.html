15void PlanarMissile::calcMotion(double tAftLaunch,double dt){
16 motion_prev=motion;
17 double V=motion.vel.norm();
18 auto propNav=getShared<PropNav>(controllers["Navigator"]);
19 Eigen::Vector3d accelCmd=propNav->commands["accel"];
20 Eigen::Vector3d omegaCmd=propNav->commands["omega"];
21 double omega=omegaCmd.norm();
22 double omegaMax=maxLoadG*gravity/V;
23 if(omega>omegaMax){
24 omegaCmd*=(omegaMax/omega);
25 omega=omegaMax;
26 accelCmd=omegaCmd.cross(motion.vel());
27 }
28 auto [accelScalar,omegaV] = calcAccel(accelCmd,tAftLaunch,dt);
29 Eigen::Vector3d ex=motion.vel()/V;
30 Eigen::Vector3d accel=ex*accelScalar;
31 omega=omegaV.norm();
32 double wt=omega*dt;
33 if(wt<1e-8){//ほぼ無回転
34 auto accelVector=motion.relBtoP(Eigen::Vector3d(accelScalar,0,0));
35 motion.setPos(motion.pos()+motion.vel()*dt+accelVector*dt*dt/2.);
36 auto posH=motion.absPtoH(motion.pos(),"NED",true);
37 posH(2)=-motion_prev.getHeight();
38 motion.setPos(motion.absHtoP(posH,"NED",true));
39 Eigen::Vector3d vn=motion.vel().normalized();
40 motion.setVel((V+accelScalar*dt)*vn);
41 auto velH=motion.velPtoH(motion.vel(),motion.pos(),"NED",true);
42 velH(2)=0;
43 motion.setVel(motion.velHtoP(velH,motion.pos(),"NED",true));
44 }else{
45 Eigen::Vector3d ex0=motion.vel()/V;
46 Eigen::Vector3d ey0=omegaV.cross(ex0).normalized();
47 Eigen::Vector3d dr=V*dt*(ex0*sinc(wt)+ey0*oneMinusCos_x2(wt)*wt)+accelScalar*dt*dt*(ex0*(sinc(wt)-oneMinusCos_x2(wt))+ey0*wt*(sincMinusOne_x2(wt)+oneMinusCos_x2(wt)));
48 motion.setPos(motion.pos()+dr);
49 auto posH=motion.absPtoH(motion.pos(),"NED",true);
50 posH(2)=-motion_prev.getHeight();
51 motion.setPos(motion.absHtoP(posH,"NED",true));
52 Eigen::Vector3d vn=(ex0*cos(wt)+ey0*sin(wt)).normalized();
53 motion.setVel((V+accelScalar*dt)*vn);
54 auto velH=motion.velPtoH(motion.vel(),motion.pos(),"NED",true);
55 velH(2)=0;
56 motion.setVel(motion.velHtoP(velH,motion.pos(),"NED",true));
57 }
58 motion.setOmega(omegaV);
59 calcQ();
60}

62std::pair<double,Eigen::Vector3d> PlanarMissile::calcAccel(const Eigen::Vector3d& accelCmd, double tAftLaunch,double dt){
63 double V=motion.vel.norm();
64 double alt=motion.getHeight();
65 bool pwroff=tAftLaunch>=tBurn;
66 double thrust_;
67 if(pwroff){
68 thrust_ = 0.0;
69 mass = massF;
70 lcg = lcgf;
71 }else{
72 thrust_ = thrust;
73 mass = massI+(massF-massI)*tAftLaunch/tBurn;
74 lcg = lcgi+(lcgf-lcgi)*tAftLaunch/tBurn;
75 }
76 std::map<std::string,double> atm=atmosphere(alt);
77 double M=V/atm["a"];
78 double qd=0.5*atm["rho"]*V*V;
79 Eigen::Vector3d ex=motion.vel()/V;
80 Eigen::Vector3d gravityAccelVector=motion.dirHtoP(Eigen::Vector3d(0,0,1),"NED")*gravity; //TODO 慣性力モデルに置換
81 Eigen::Vector3d ey=(accelCmd-gravityAccelVector).cross(ex)*mass;
82 double desiredSideForce=ey.norm();
83 Eigen::Vector3d sideAx;
84 double aoa,CL,CD;
85 if(desiredSideForce>0){
86 ey/=desiredSideForce;
87 sideAx=ex.cross(ey);
88 std::pair<double,double> tmp=EkkerMDT::cldcmd(d,l,ln,lcg,bt,St,M);
89 double cld=tmp.first;
90 double cmd=tmp.second;
91 auto func=[&](double aoa_){
92 std::pair<double,double> tmp=EkkerMDT::clacma(d,l,ln,lcg,bw,Sw,lw,bt,St,M,aoa_);
93 double cla=tmp.first;
94 double cma=tmp.second;
95 double delta=std::clamp<double>(-cma/cmd*aoa_,-maxD,maxD);
96 return (cla*aoa_+cld*delta)*(qd*Sref)+thrust_*sin(aoa_)-desiredSideForce;
97 };
98 if(func(0.0)>=0){
99 aoa=0.0;
100 }else if(func(maxA)<=0){
101 aoa=maxA;
102 }else{
103 try{
104 boost::math::tools::eps_tolerance<double> tol(16);
105 boost::uintmax_t max_iter=10;
106 auto result=boost::math::tools::toms748_solve(func,0.0,maxA,tol,max_iter);
107 aoa=(result.first+result.second)/2;
108 }catch(std::exception& e){
109 std::cout<<"exception at PlanarMissile::calcMotion"<<std::endl;
110 DEBUG_PRINT_EXPRESSION(func(0.0))
111 DEBUG_PRINT_EXPRESSION(func(maxA))
112 DEBUG_PRINT_EXPRESSION(alt)
113 DEBUG_PRINT_EXPRESSION(M)
114 DEBUG_PRINT_EXPRESSION(desiredSideForce)
115 DEBUG_PRINT_EXPRESSION(thrust_)
116 throw e;
117 }
118 }
119 tmp=EkkerMDT::clacma(d,l,ln,lcg,bw,Sw,lw,bt,St,M,aoa);
120 double cla=tmp.first;
121 double cma=tmp.second;
122 double delta=std::clamp<double>(-cma/cmd*aoa,-maxD,maxD);
123 //CL=(cla-cld*cma/cmd)*aoa;
124 CL=cla*aoa+cld*delta;
125 CD=EkkerMDT::cdtrim(d,l,ln,bw,Sw,thicknessRatio,bt,St,thicknessRatio,M,alt,aoa,delta,frictionFactor,pwroff);
126 }else{
127 sideAx<<0,0,0;
128 CL=0.0;
129 CD=EkkerMDT::cdtrim(d,l,ln,bw,Sw,thicknessRatio,bt,St,thicknessRatio,M,alt,0,0,frictionFactor,pwroff);
130 aoa=0.0;
131 }
132 Eigen::Vector3d sideAccel=sideAx*(CL*qd*Sref+thrust_*sin(aoa))/mass+(gravityAccelVector-ex*(gravityAccelVector.dot(ex)));
133 sideAccel=motion.relPtoH(sideAccel,motion.pos(),"NED",true);
134 sideAccel(2)=0;
135 sideAccel=motion.relHtoP(sideAccel,motion.pos(),"NED",true);
136 //attitude and deflection are assumed to be immidiately adjustable as desired.
137 double accelScalar=(-CD*Sref*qd+thrust_*cos(aoa))/mass+gravityAccelVector.dot(ex);
138 Eigen::Vector3d omegaV=motion.vel().cross(sideAccel)/(V*V);
139 return {accelScalar,omegaV};
140}

142bool PlanarMissile::calcRangeSub(double vs,double hs,double vt,double ht,double obs,double aa,double r){
143 Eigen::Vector3d posBef=motion.pos();
144 Eigen::Vector3d velBef=motion.vel();
145 Eigen::Vector3d omegaBef=motion.omega();
146 Quaternion qBef=motion.q;
147 Time timeBef=motion.time;
148 MotionState motion_prevBef=motion_prev;
149 bool isBoundToParentBef=isBoundToParent;
150 if(isBoundToParentBef){
151 unbindFromParent();
152 }
153 double dt=interval[SimPhase::BEHAVE]*manager->getBaseTimeStep();
154 motion.setPos(motion.absHtoP(Eigen::Vector3d(0,0,-hs),"NED",true));
155 motion.setVel(motion.velHtoP(Eigen::Vector3d(cos(obs),-sin(obs),0)*std::max(vs,1e-3),Eigen::Vector3d(0,0,-hs),"NED",true));
156 motion.setOmega(Eigen::Vector3d(0,0,0));
157 calcQ();
158 Eigen::Vector3d tpos=motion.absHtoP(Eigen::Vector3d(r,0,-hs),"NED",true);
159 Eigen::Vector3d tvel=motion.velHtoP(Eigen::Vector3d(cos(aa),sin(aa),0)*vt,Eigen::Vector3d(r,0,-hs),"NED",true);
160 Eigen::Vector3d tpos_prev=tpos-dt*tvel;
161 motion_prev.setPos(motion.pos()-dt*motion.vel());
162 motion_prev.setVel(motion.vel());
163 double t=0;
164 bool finished=false;
165 bool hit=false;
166 auto propNav=getShared<PropNav>(controllers["Navigator"]);
167 bool hasLaunchedBef=hasLaunched;
168 hasLaunched=true;
169 while(!finished){
170 commands={
171 {"Navigator",{
172 {"rs",motion.pos()},
173 {"vs",motion.vel()},
174 {"rt",tpos},
175 {"vt",tvel}
176 }}
177 };
178 propNav->control();
179 calcMotion(t,dt);
180 tpos_prev=tpos;
181 tpos+=dt*tvel;
182 t+=dt;
183 if(hitCheck(tpos,tpos_prev)){
184 finished=true;
185 hit=true;
186 }else if(checkDeactivateCondition(t)){
187 finished=true;
188 hit=false;
189 }
190 }
191 if(isBoundToParentBef){
192 bindToParent();
193 }
194 motion.setPos(posBef);
195 motion.setVel(velBef);
196 motion.setOmega(omegaBef);
197 motion.setTime(timeBef);
198 motion.setQ(qBef);
199 motion_prev=motion_prevBef;
200 hasLaunched=hasLaunchedBef;
201 return hit;
202}

204void PlanarMissile::makeRangeTable(const std::string& dstPath){
205 py::gil_scoped_acquire acquire;
206 std::cout<<"makeRangeTable start"<<std::endl;
207 int nvs=7;
208 int nhs=6;
209 int nvt=7;
210 int nht=2;//
211 int nobs=7;
212 int naa=13;
213 Eigen::VectorXd vs=Eigen::VectorXd::LinSpaced(nvs,0.0,600.0);
214 Eigen::VectorXd hs=Eigen::VectorXd::LinSpaced(nhs,0.0,20000.0);
215 Eigen::VectorXd vt=Eigen::VectorXd::LinSpaced(nvt,0.0,600.0);
216 Eigen::VectorXd ht=Eigen::VectorXd::LinSpaced(nht,0.0,20000.0);
217 Eigen::VectorXd obs=Eigen::VectorXd::LinSpaced(nobs,0.0,M_PI);
218 Eigen::VectorXd aa=Eigen::VectorXd::LinSpaced(naa,-M_PI,M_PI);
219 int numDataPoints=nvs*nhs*nht*nvt*nobs*naa;
220 // RowMajorは最後の軸に沿って連続、最初の軸が最外周
221 // ColMajorは最初の軸に沿って連続、最後の軸が最外周
222 // EigenはColMajorがデフォルトだが、テーブルデータとしてはRowMajorの方が都合がよいため、RowMajorのテーブルが得られるようにデータ点を並べる。
223 Eigen::MatrixXd args=Eigen::MatrixXd::Zero(6,numDataPoints);//ここはColMajorでよい。
224 std::cout<<"create indices..."<<std::endl;
225 int idx=0;
226 for(int vs_i=0;vs_i<nvs;++vs_i){
227 for(int hs_i=0;hs_i<nhs;hs_i++){
228 for(int vt_i=0;vt_i<nvt;vt_i++){
229 for(int ht_i=0;ht_i<nht;ht_i++){
230 for(int obs_i=0;obs_i<nobs;obs_i++){
231 for(int aa_i=0;aa_i<naa;aa_i++){
232 args.block(0,idx,6,1)<<vs(vs_i),hs(hs_i),vt(vt_i),ht(ht_i),obs(obs_i),aa(aa_i);//ここはColMajorでよい。
233 idx++;
234 }
235 }
236 }
237 }
238 }
239 }
240 std::cout<<"create indices done."<<std::endl;
241 int numProcess=std::thread::hardware_concurrency();
242 std::vector<std::thread> th;
243 std::vector<std::future<Eigen::VectorXd>> f;
244 std::vector<int> begins;
245 int numPerProc=numDataPoints/numProcess;
246 std::cout<<"run in "<<numProcess<<" processes. total="<<numDataPoints<<", perProc="<<numPerProc<<std::endl;
247 std::mutex mtx;
248 std::vector<std::shared_ptr<Missile>> msls;
249 for(int i=0;i<numProcess;++i){
250 nl::json dummy_ic=instanceConfig;
251 dummy_ic["removeWhenKilled"]=true;
252 dummy_ic["entityFullName"]=getFullName()+"/for_range_table_"+std::to_string(i+1);
253 msls.push_back(createEntityByClassName<Missile>(isEpisodic(),"PhysicalAsset",getFactoryClassName(),modelConfig,dummy_ic));
254 std::promise<Eigen::VectorXd> p;
255 f.emplace_back(p.get_future());
256 Eigen::MatrixXd subargs=args.block(0,numPerProc*i,6,(i!=numProcess-1 ? numPerProc : numDataPoints-numPerProc*i));
257 th.emplace_back(makeRangeTableSub,std::ref(mtx),std::move(p),msls[i],subargs);
258 }
259 std::cout<<"waiting calculation"<<std::endl;
260 std::vector<Eigen::VectorXd> rangeTablePoints_={vs,hs,vt,ht,obs,aa};
261 Eigen::VectorXd returned(numDataPoints);
262 for(int i=0;i<numProcess;++i){
263 try{
264 returned.block(numPerProc*i,0,(i!=numProcess-1 ? numPerProc : numDataPoints-numPerProc*i),1)=f[i].get();
265 }catch(std::exception& ex){
266 {
267 std::lock_guard<std::mutex> lock(mtx);
268 std::cout<<"exception in proc("<<i<<"): "<<ex.what()<<std::endl;
269 }
270 }
271 }
272 Eigen::Tensor<double,6,Eigen::RowMajor> rangeTable_=Eigen::TensorMap<Eigen::Tensor<double,6,Eigen::RowMajor>>(returned.data(),nvs,nhs,nvt,nht,nobs,naa);
273 for(int i=0;i<numProcess;++i){
274 th[i].join();
275 }
276 for(auto&& msl: msls){
277 manager->requestToKillAsset(msl);
278 }
279 std::cout<<"exporting to npz file"<<std::endl;
280 auto np=py::module_::import("numpy");
281 np.attr("savez")(dstPath,py::arg("vs")=vs,py::arg("hs")=hs,py::arg("vt")=vt,py::arg("ht")=ht,py::arg("obs")=obs,py::arg("aa")=aa,py::arg("ranges")=rangeTable_);
282 std::cout<<"makeRangeTable done."<<std::endl;
283}
