213nl::json SimulationManager::parseConfig(const nl::json& config_){
214 nl::json ret=nl::json::object();
215 if(config_.is_object()){
216 ret=config_;
217 }else if(config_.is_array()){
218 for(auto& e:config_){
219 if(e.is_object()){
220 ret.merge_patch(e);
221 }else if(e.is_string()){
222 std::ifstream ifs(e.get<std::string>());
223 nl::json sub;

848std::uint64_t SimulationManager::getPossibleNextTickCount(const SimPhase& phase) const{
849 // 現在の処理状況(tickCountと直近のphase)において、
850 // 引数で与えられたphaseが次に実行される可能性のある時刻(tick)を返す。
851 // phaseの処理中には呼び出さないこと。
852 switch(phase){
853 case SimPhase::PERCEIVE:
854 //inReset=trueを除くとつねにtickCount+1
855 return tickCount+1;
856 case SimPhase::CONTROL:
857 if(lastAssetPhase==SimPhase::CONTROL || lastAssetPhase==SimPhase::BEHAVE){
858 return tickCount+1;
859 }else{
860 //NONE,VALIDATE,PERCEIVE
861 return tickCount;
862 }
863 case SimPhase::BEHAVE:
864 if(lastAssetPhase==SimPhase::BEHAVE){
865 return tickCount+1;
866 }else{
867 //NONE,VALIDATE,PERCEIVE,CONTROL
868 return tickCount;
869 }
870 case SimPhase::AGENT_STEP:
871 if(lastAssetPhase==SimPhase::CONTROL || lastAssetPhase==SimPhase::BEHAVE){
872 return tickCount+1;
873 }else{
874 //NONE,VALIDATE,PERCEIVE
875 return tickCount;
876 }
877 case SimPhase::ON_INNERSTEP_BEGIN:
878 if(lastAssetPhase==SimPhase::PERCEIVE){
879 if(lastGroup1CallbackPhase==SimPhase::ON_INNERSTEP_BEGIN){
880 return tickCount+1;
881 }else{
882 return tickCount;
883 }
884 }else{
885 return tickCount+1;
886 }
887 case SimPhase::ON_INNERSTEP_END:
888 if(lastAssetPhase==SimPhase::PERCEIVE){
889 // onInnerStepEndはつねにonInnerStepBeginと対になって実行される
890 if(lastGroup1CallbackPhase==SimPhase::ON_INNERSTEP_BEGIN){
891 return tickCount;
892 }else{
893 return tickCount+1;
894 }
895 }else{
896 return tickCount+1;
897 }
898 default: // non-periodic phase
899 return tickCount;
900 }

1703void SimulationManager::runAssetPhaseFunc(const SimPhase& phase){
1704 //後の処理で各phaseごとの関数呼び出し形式を共通化しておく
1705 static const std::map<SimPhase,std::function<void(const std::shared_ptr<Asset>&)>> phaseFunc={
1706 {SimPhase::PERCEIVE,[](const std::shared_ptr<Asset>& a){a->perceive(false);}},
1707 {SimPhase::CONTROL,[](const std::shared_ptr<Asset>& a){a->control();}},
1708 {SimPhase::BEHAVE,[](const std::shared_ptr<Asset>& a){a->behave();}}
1709 };
1710 assert(phaseFunc.count(phase)>0);
1711 //phaseFunc実行中のエラー監視用フラグ。子スレッドからの例外送出を拾わないため必要。
1712 std::atomic_bool success=true;
1713
1714 updateProcessQueueIfNeeded(phase);
1715 //この時刻のこのphaseで処理すべきAssetの範囲を取得する
1716 auto itr = asset_process_queue[phase].find(tickCount);
1717 if(itr==asset_process_queue[phase].end()){
1718 //この時刻の処理対象なし。
1719 calcNextTick(phase);
1720 return;
1721 }
1722 auto toBeProcessed = std::move(itr->second);
1723 asset_process_queue[phase].erase(itr);
1724 if(numThreads==1){
1725 //シングルスレッドの場合、先頭から順番に処理すればよい
1726 std::chrono::system_clock::time_point before,after;
1727 std::chrono::microseconds difference;
1728 std::string phaseName=enumToStr(phase);
1729 for(auto&& [order,asset] : toBeProcessed){
1730 std::string name=asset->getFullName();
1731 if(asset->isAlive()){
1732 try{
1733 if(measureTime){
1734 before=std::chrono::system_clock::now();
1735 }
1736 phaseFunc.at(phase)(asset);
1737 if(measureTime){
1738 after=std::chrono::system_clock::now();
1739 difference=std::chrono::duration_cast<std::chrono::microseconds>(after-before);
1740 maxProcessTime[phaseName][name]=std::max<std::uint64_t>(maxProcessTime[phaseName][name],difference.count());
1741 minProcessTime[phaseName][name]=std::min<std::uint64_t>(minProcessTime[phaseName][name],difference.count());
1742 processCount[phaseName][name]++;
1743 meanProcessTime[phaseName][name]+=((double)(difference.count())-meanProcessTime[phaseName][name])/processCount[phaseName][name];
1744 }
1745 }catch(std::exception& ex){
1746 std::cout<<"asset."<<phaseName<<"() failed. asset="<<name<<std::endl;
1747 std::cout<<ex.what()<<std::endl;
1748 std::cout<<"w="<<worker_index()<<","<<"v="<<vector_index()<<std::endl;
1749 success=false;
1750 throw ex;
1751 }
1752 }
1753 }
1754 }else{
1755 //並列化する場合、処理できるようになったものから処理を実施
1756 py::gil_scoped_release release;
1757 std::set<std::size_t> waitings;
1758 for(auto&& index : toBeProcessed | std::views::keys){
1759 waitings.insert(index);
1760 }
1761
1762 //処理状況監視用の変数の初期化
1763 std::map<std::size_t,std::atomic_bool> futures;//各Assetの処理状況を格納する。終了時にtrueとなる。
1764 std::map<std::size_t,std::vector<std::size_t>> checklists;//各Assetが依存するAssetのindexリスト。trueになった後に何度も確認してしまわないように要素を削除しながら回す。
1765 for(auto&& index : waitings){
1766 futures[index]=false;
1767 checklists[index];
1768 for(auto&& predecessor : asset_dependency_graph[phase].get_predecessors_by_index(index)){
1769 if(toBeProcessed.contains(predecessor)){
1770 checklists[index].push_back(predecessor);
1771 }
1772 }
1773 }
1774
1775 //処理を順番に実行する
1776 auto wit=waitings.begin();
1777 while(wit!=waitings.end()){
1778 std::size_t index=*wit;
1779 bool ready=true;
1780 auto dit=checklists[index].begin();
1781 while(dit!=checklists[index].end()){
1782 if(futures.find(*dit)==futures.end() || futures[*dit]){
1783 dit=checklists[index].erase(dit);
1784 }else{
1785 ready=false;
1786 dit=checklists[index].end();
1787 }
1788 }
1789 if(ready){
1790 auto asset=toBeProcessed.at(index);
1791 std::string name=asset->getFullName();
1792 if(asset->isAlive()){
1793 int w=worker_index();
1794 int v=vector_index();
1795 pool.detach_task([&futures,&success,phase,index,name,asset,w,v,this]{
1796 std::chrono::system_clock::time_point before,after;
1797 std::chrono::microseconds difference;
1798 try{
1799 if(measureTime){
1800 before=std::chrono::system_clock::now();
1801 }
1802 phaseFunc.at(phase)(asset);
1803 if(measureTime){
1804 std::string phaseName=enumToStr(phase);
1805 after=std::chrono::system_clock::now();
1806 difference=std::chrono::duration_cast<std::chrono::microseconds>(after-before);
1807 maxProcessTime.at(phaseName).at(name)=std::max<std::uint64_t>(maxProcessTime.at(phaseName).at(name),difference.count());
1808 minProcessTime.at(phaseName).at(name)=std::min<std::uint64_t>(minProcessTime.at(phaseName).at(name),difference.count());
1809 processCount.at(phaseName).at(name)++;
1810 meanProcessTime.at(phaseName).at(name)+=((double)(difference.count())-meanProcessTime.at(phaseName).at(name))/processCount.at(phaseName).at(name);
1811 }
1812 }catch(std::exception& ex){
1813 std::cout<<"asset."<<enumToStr(phase)<<"() failed. asset="<<name<<std::endl;
1814 std::cout<<ex.what()<<std::endl;
1815 std::cout<<"w="<<w<<","<<"v="<<v<<std::endl;
1816 success=false;
1817 throw ex;
1818 }
1819 futures[index]=true;
1820 });
1821 }else{
1822 //not alive
1823 futures[index]=true;
1824 }
1825 wit=waitings.erase(wit);
1826 }else{
1827 ++wit;
1828 }
1829 if(wit==waitings.end()){
1830 wit=waitings.begin();
1831 }
1832 if(!success){
1833 throw std::runtime_error("SimulationManager::runAssetPhaseFunc("+enumToStr(phase)+") failed.");
1834 }
1835 }
1836 pool.wait();
1837 }
1838 lastAssetPhase=phase;
1839 killAndRemoveAssets();
1840 setupNewEntities();
1841 //次回処理時刻の算出
1842 for(auto&& [index,asset] : toBeProcessed){
1843 calcNextTick(phase,asset);
1844 }
1845 calcNextTick(phase);
1846}

1851 {SimPhase::ON_MAKE_OBS,[](const std::shared_ptr<Callback>& c){c->onMakeObs();}},
1852 {SimPhase::ON_DEPLOY_ACTION,[](const std::shared_ptr<Callback>& c){c->onDeployAction();}},
1853 {SimPhase::ON_EPISODE_BEGIN,[](const std::shared_ptr<Callback>& c){c->onEpisodeBegin();}},
1854 {SimPhase::ON_VALIDATION_END,[](const std::shared_ptr<Callback>& c){c->onValidationEnd();}},
1855 {SimPhase::ON_STEP_BEGIN,[](const std::shared_ptr<Callback>& c){c->onStepBegin();}},

1859 {SimPhase::ON_EPISODE_END,[](const std::shared_ptr<Callback>& c){c->onEpisodeEnd();}}

1864 std::chrono::system_clock::time_point before,after;
1865 std::chrono::microseconds difference;
1866 std::string phaseName=enumToStr(phase);
1867 if(phase==SimPhase::ON_GET_OBSERVATION_SPACE ||
1868 phase==SimPhase::ON_GET_ACTION_SPACE ||
1869 phase==SimPhase::ON_MAKE_OBS ||
1870 phase==SimPhase::ON_DEPLOY_ACTION ||
1871 phase==SimPhase::ON_EPISODE_BEGIN ||
1872 phase==SimPhase::ON_EPISODE_END ||
1873 phase==SimPhase::ON_VALIDATION_END ||

1878 for(auto&& callback : callback_dependency_graph[phase].get_range()){
1879 if(group1 != (isinstance<Ruler>(callback) || isinstance<Reward>(callback))){
1880 continue;

1882 std::string name=callback->getFullName();
1883 try{
1884 if(measureTime){
1885 before=std::chrono::system_clock::now();
1886 }
1887 phaseFunc.at(phase)(callback);
1888 if(measureTime){
1889 after=std::chrono::system_clock::now();
