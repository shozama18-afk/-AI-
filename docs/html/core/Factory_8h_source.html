78 /**
79 * @brief C++クラスを追加する。
80 *
81 * @tparam Class 対象のC++クラス
82 * @param [in] baseName 追加先のグループ名(base name)
83 * @param [in] className クラス名
84 */
85 template<typename Class>
86 static void addClass(const std::string& baseName,const std::string& className){
87 if constexpr (std::is_abstract_v<Class>){
88 // 抽象クラスは例外を投げる関数を登録しておく。
89 registerConstructor(baseName,className,[](const nl::json& modelConfig,const nl::json& instanceConfig) -> std::shared_ptr<Entity> {
90 throw std::runtime_error(cereal::util::demangle(typeid(Class).name())+" is abstract class.");
91 });
92 }else{
93 registerConstructor(baseName,className,&Class::template create<typename Class::Type,const nl::json&,const nl::json&>);
94 }
95 registerIsinstanceFunc(baseName,className,[](const std::shared_ptr<const Entity>& instance){
96 return isinstance<Class>(instance);
97 });
98 }
