140void PhysicalAsset::setInitialMotionStateFromJson(const nl::json& j){
141 if(j.contains("motion")){
142 // MotionStateを直接指定する場合
143 motion.load_from_json(j.at("motion"));
144
145 //Body座標軸の修正
146 if(j.contains("bodyAxisOrder")){
147 bodyAxisOrder=j.at("bodyAxisOrder");
148 }else if(modelConfig.contains("bodyAxisOrder")){
149 bodyAxisOrder=modelConfig.at("bodyAxisOrder");
150 }else{
151 bodyAxisOrder=motion.axisOrder;
152 }
153 motion.setAxisOrder(bodyAxisOrder);
154
155 //親座標系の修正
156 std::shared_ptr<CoordinateReferenceSystem> parentCRS;
157 if(isBoundToParent){
158 //親に固定されている場合はそのBody座標系でMotionStateを管理する。
159 if(parent.expired()){
160 throw std::runtime_error("The parent of '"+getFullName()+"' was removed without removing its bound child.");
161 }
162 parentCRS=parent.lock()->getBodyCRS();
163 }else{
164 //親に固定されていない場合はSimulationManagerの基準座標系でMotionStateを管理する。
165 parentCRS=manager->getRootCRS();
166 }
167 motion.setCRS(parentCRS,true);
168 }else{
169 // 各状態量を個別に指定する場合
170 if(j.contains("bodyAxisOrder")){
171 bodyAxisOrder=j.at("bodyAxisOrder");
172 }else if(modelConfig.contains("bodyAxisOrder")){
173 bodyAxisOrder=modelConfig.at("bodyAxisOrder");
174 }else{
175 bodyAxisOrder="FSD";
176 }
177 motion.setTime(manager->getTime());
178 motion.setAxisOrder(bodyAxisOrder);
179
180 //初期条件を計算するための中間座標系の設定
181 if(j.contains("crs")){
182 //instanceConfigで指定した場合はその座標系を使用する
183 auto initialStateCRS=createOrGetEntity<CoordinateReferenceSystem>(j.at("crs"));
184 motion.setCRS(initialStateCRS,false);
185 }else{
186 //初期条件の座標系を指定しなかった場合は、親があれば親のBody座標系、無ければRulerのlocal座標系とみなす
187 if(parent.expired()){
188 motion.setCRS(manager->getRuler().lock()->getLocalCRS(),false);
189 }else{
190 motion.setCRS(parent.lock()->getBodyCRS(),false);
191 }
192 }
193 // ----座標計算ここから----
194 // 位置の設定(省略時は中間座標系の原点)
195 if(j.contains("pos")){
196 nl::json pos_j=getValueFromJsonKR(j,"pos",randomGen);
197 Eigen::Vector3d pos_v;
198 if(pos_j.is_array() && pos_j.size()==3){
199 pos_v=pos_j;
200 }else{
201 throw std::runtime_error("PhysicalAsset::setInitialMotionStateFromJson() failed. 'pos' should be 3-D vector.");
202 }
203 motion.setPos(pos_v);
204 }
205
206 // 姿勢の計算
207 // 中間座標系のXYZ軸から方位角(ヨー)、仰角(ピッチ)、バンク角(ロール)の順に回転させてBody座標系を得るものとする。
208 // 中間座標系がNED座標の場合、一般に使用される方位角、仰角、バンク角と一致する。
209
210 // 方位角の設定
211 double azInDeg=0.0;
212 if(j.contains("heading")){
213 azInDeg=getValueFromJsonKR(j,"heading",randomGen);
214 }else if(j.contains("azimuth")){
215 azInDeg=getValueFromJsonKR(j,"azimuth",randomGen);
216 }else if(j.contains("az")){
217 azInDeg=getValueFromJsonKR(j,"az",randomGen);
218 }
219 while(azInDeg>180){azInDeg-=360;}
220 while(azInDeg<=-180){azInDeg+=360;}
221 double az=deg2rad(azInDeg);
222 double cs=cos(az);
223 double sn=sin(az);
224 double cs_2=cos(az/2);
225 double sn_2=sin(az/2);
226 if(azInDeg==0){
227 cs=1;sn=0;
228 cs_2=1;sn_2=0;
229 }else if(azInDeg==90){
230 cs=0;sn=1;
231 cs_2=sqrt(2)/2;
232 sn_2=sqrt(2)/2;
233 }else if(azInDeg==180){
234 cs=-1;sn=0;
235 cs_2=0;sn_2=1;
236 }else if(azInDeg==-90){
237 cs=0;sn=-1;
238 cs_2=sqrt(2)/2;
239 sn_2=-sqrt(2)/2;
240 }
241 // 仰角の設定
242 double elInDeg=0.0;
243 if(j.contains("pitch")){
244 elInDeg=getValueFromJsonKR(j,"pitch",randomGen);
245 }else if(j.contains("elevation")){
246 elInDeg=getValueFromJsonKR(j,"elevation",randomGen);
247 }else if(j.contains("el")){
248 elInDeg=getValueFromJsonKR(j,"el",randomGen);
249 }
250 while(elInDeg>180){elInDeg-=360;}
251 while(elInDeg<=-180){elInDeg+=360;}
252 double el=deg2rad(elInDeg);
253 // バンク角の設定
254 double rollInDeg=0.0;
255 if(j.contains("roll")){
256 rollInDeg=getValueFromJsonKR(j,"roll",randomGen);
257 }else if(j.contains("bank")){
258 rollInDeg=getValueFromJsonKR(j,"bank",randomGen);
259 }
260 while(rollInDeg>180){rollInDeg-=360;}
261 while(rollInDeg<=-180){rollInDeg+=360;}
262 double roll=deg2rad(rollInDeg);
263
264 Quaternion q1=Quaternion::fromAngle(Eigen::Vector3d(0,0,1),az);
265 Quaternion q2=Quaternion::fromAngle(Eigen::Vector3d(0,1,0),el);
266 Quaternion q3=Quaternion::fromAngle(Eigen::Vector3d(1,0,0),roll);
267 Quaternion qToXYZFromFSD = q1*q2*q3;
268 Quaternion qToXYZFromBody = qToXYZFromFSD*getAxisSwapQuaternion("FSD",bodyAxisOrder); // [XYZ <- Body]
269 motion.setQ(qToXYZFromBody); // [Parent <- Body]
270
271 // 速度の設定(省略時は中間座標系において静止)
272 Eigen::Vector3d vel_v;
273 if(j.contains("vel")){
274 nl::json vel_j=getValueFromJsonKR(j,"vel",randomGen);
275 if(vel_j.is_array() && vel_j.size()==3){
276 vel_v=vel_j;
277 }else if(vel_j.is_number()){
278 double V=vel_j;
279 double alphaInDeg=getValueFromJsonKRD(j,"alpha",randomGen,0.0);
280 while(alphaInDeg>180){alphaInDeg-=360;}
281 while(alphaInDeg<=-180){alphaInDeg+=360;}
282 double alpha=deg2rad(alphaInDeg);
283 double betaInDeg=getValueFromJsonKRD(j,"beta",randomGen,0.0);
284 while(betaInDeg>180){betaInDeg-=360;}
285 while(betaInDeg<=-180){betaInDeg+=360;}
286 double beta=deg2rad(betaInDeg);
287 Eigen::Vector3d velInFSD=V*Eigen::Vector3d(cos(alpha)*cos(beta),sin(beta),sin(alpha)*cos(beta));
288 vel_v=qToXYZFromFSD.transformVector(velInFSD);
289 }else{
290 throw std::runtime_error("PhysicalAsset::setInitialMotionStateFromJson() failed. 'vel' should be 3-D vector or scalar value.");
291 }
292 }else{
293 vel_v=Eigen::Vector3d::Zero();
294 }
295 motion.setVel(vel_v);
296
297 // 角速度の設定(省略時は中間座標系において静止)
298 Eigen::Vector3d omega_v;
299 if(j.contains("omegaB")){
300 // Body座標系からみた値で指定された場合
301 omega_v=getValueFromJsonKR(j,"omegaB",randomGen);
302 omega_v=qToXYZFromBody.transformVector(omega_v);
303 }else if(j.contains("omegaP")){
304 // 中間座標系からみた値で指定された場合
305 omega_v=getValueFromJsonKR(j,"omegaP",randomGen);
306 }else{
307 omega_v=Eigen::Vector3d::Zero();
308 }
309 motion.setOmega(omega_v);
310 // ----座標計算ここまで----
311
312 //親座標系の修正
313 std::shared_ptr<CoordinateReferenceSystem> parentCRS;
314 if(isBoundToParent){
315 //親に固定されている場合はそのBody座標系でMotionStateを管理する。
316 if(parent.expired()){
317 throw std::runtime_error("The parent of '"+getFullName()+"' was removed without removing its bound child.");
318 }
319 parentCRS=parent.lock()->getBodyCRS();
320 }else{
321 //親に固定されていない場合はSimulationManagerの基準座標系でMotionStateを管理する。
322 parentCRS=manager->getRootCRS();
323 }
324 motion.setCRS(parentCRS,true);
325 }
326 if(bodyCRS){
327 bodyCRS->setBaseCRS(motion.getCRS());
328 bodyCRS->updateByMotionState(motion);
329 }
330 motion_prev=motion.extrapolate(-getInterval(SimPhase::BEHAVE)*manager->getBaseTimeStep());
331}
