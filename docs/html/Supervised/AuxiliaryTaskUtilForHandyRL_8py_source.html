108 def add_episodes(self,episodes):
109 ep_cnt = 0
110 for ep in episodes:
111 extracted=self.extract_data(ep)
112 if(extracted is None):
113 continue
114 self.episodes.append(extracted)
115 num_candidates = 1 + max(0, ep['steps'] - self.args['forward_steps'])
116 for train_st in range(num_candidates):
117 ep_idx = self._ep_idx_ofs + len(self.episodes) - 1
118 self.ep_indices[self._next_idx]=(ep_idx, train_st)
119 self.flat_idx_from_ep[ep_idx][train_st] = self._next_idx
120 self._next_idx += 1
121 self._data_count = min(self._data_count + 1, self.capacity)
122 self._total_data_count += 1
123 if self._next_idx >= self.capacity:
124 self._next_idx = 0
125 ep_cnt += 1
126 while self.ep_indices[self._next_idx][0] > self._ep_idx_ofs:
127 self.flat_idx_from_ep.pop(self._ep_idx_ofs)
128 self._ep_idx_ofs += 1
129 self.episodes.popleft()
130 def get_episode(self,ep_idx):
131 return self.episodes[ep_idx-self._ep_idx_ofs]
132 def get_idx_candidates(self):

149 def make_batch(self,task_name,episodes):
150 inputs_tot=[]
151 truth_tot=[]
152 mask_tot=[]
153 for ep in episodes:
154 key = task_name if task_name in ep['inputs'] else 'original'
155 inputs = ep['inputs'][key][:ep['end'] - ep['start']]
156 if task_name in self.truth_zeros:
157 truth = ep['truths'][task_name][:ep['end'] - ep['start']]
158 mask = ep['mask'][:ep['end'] - ep['start']]
159
160 # pad each array if step length is short
161 batch_steps = self.args['burn_in_steps'] + self.args['forward_steps']
162 if len(mask) < batch_steps:
163 pad_len_b = self.args['burn_in_steps'] - (ep['train_start'] - ep['start'])
164 pad_len_a = batch_steps - len(mask) - pad_len_b
165 inputs = map_r(inputs, lambda i: np.pad(i, [(pad_len_b, pad_len_a)] + [(0, 0)] * (len(i.shape) - 1), 'constant', constant_values=0))
166 if task_name in self.truth_zeros:
167 truth = map_r(truth, lambda t: np.pad(t, [(pad_len_b, pad_len_a)] + [(0, 0)] * (len(t.shape) - 1), 'constant', constant_values=0))
168 mask = np.pad(mask, [(pad_len_b, pad_len_a), (0, 0), (0, 0)], 'constant', constant_values=0)
169 inputs_tot.append(inputs)
170 if task_name in self.truth_zeros:
171 truth_tot.append(truth)
172 mask_tot.append(mask)
173 inputs = to_torch(bimap_r(self.inputs_zeros[key],rotate(inputs_tot), lambda _, i: np.array(i)))
174 if task_name in self.truth_zeros:
175 truth = to_torch(bimap_r(self.truth_zeros[key],rotate(truth_tot), lambda _, t: np.array(t)))
176 mask = to_torch(np.array(mask_tot))
177 return {
178 'inputs':inputs,
179 'truth':truth,
180 'mask':mask

182 def forward(self,model,batch):
183 inputs = batch['inputs']
184 batch_shape = batch['mask'].size()[:3] # (B, T, P or 1)
185 hidden = batch.get('hidden_in',None)
186 if hidden is None:
187 # feed-forward neural network
188 inputs = map_r(inputs, lambda i: i.flatten(0, 2)) # (..., B * T * P or 1, ...)
189 outputs = model(inputs, None)
190 outputs = map_r(outputs, lambda o: o.unflatten(0, batch_shape)) # (..., B, T, P or 1, ...)
191 else:
192 # sequential computation with RNN
193 # RNN block is assumed to be an instance of ASRCAISim1.plugins.HandyRLUtility.RecurrentBlock.RecurrentBlock
194 outputs=[]
195 seq_len = batch_shape[1]
196 hidden = map_r(hidden, lambda h: h.unsqueeze(0).unsqueeze(0).expand((batch_shape[0],batch_shape[2])+h.shape)) # (..., B, P or 1, ...)
197 inputs = map_r(inputs, lambda i: i.transpose(1, 2)) # (..., B , P or 1 , T, ...)
198 mask_ = map_r(batch['mask'], lambda m: m.transpose(1, 2)) # (..., B , P or 1 , T, ...)
199 mask = map_r(hidden, lambda h: mask_.view([*h.size()[:2], seq_len, *([1] * (h.dim() - 2))]).flatten(0,1))
200 hidden_ = map_r(hidden, lambda h: h.flatten(0, 1)) # (..., B * P or 1, ...)
201 if self.args['burn_in_steps'] >0 :
202 model.eval()
203 with torch.no_grad():
204 outputs_, hidden_ = model(map_r(inputs, lambda i:i[:, :, :self.args['burn_in_steps'], ...].flatten(0, 2)), hidden_, self.args['burn_in_steps'],mask=map_r(mask,lambda m: m[:,:self.args['burn_in_steps'],...]))
205 outputs_ = map_r(outputs_, lambda o: o.unflatten(0, (batch_shape[0], batch_shape[2], self.args['burn_in_steps'])).transpose(1, 2))
206 outputs.append(outputs_)
207 if not model.training:
208 model.train()
209 outputs_, hidden_ = model(map_r(inputs, lambda i:i[:, :, self.args['burn_in_steps']:, ...].flatten(0, 2)), hidden_,seq_len-self.args['burn_in_steps'],mask=map_r(mask,lambda m: m[:,self.args['burn_in_steps']:,...]))
210 outputs_ = map_r(outputs_, lambda o: o.unflatten(0, (batch_shape[0], batch_shape[2], seq_len-self.args['burn_in_steps'])).transpose(1, 2))
211 outputs.append(outputs_)
212 outputs = torch.cat(outputs, dim=1).mul(batch['mask'])
213 return outputs
