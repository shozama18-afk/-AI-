30 def getPolicyNameToDisplay(self, frame):
31 """各陣営のAgentを表す名称を取得するもの。
32 R8年度コンテストでは護衛対象機も独立したAgentを持っているため、それを除いて先頭に来るもののpolicy名を用いる。
33
34 Args:
35 frame (dict): 現在のフレームデータ
36
37 Returns:
38 str: Agent名表示用の文字列
39 """
40 ruler=frame["ruler"]
41 policyName = {}
42 for team in [ruler["westSider"], ruler["eastSider"]]:
43 policyName[team]=""
44 for agentFullName,agent in frame["agents"].items():
45 if agent["team"]==team:
46 name,model,policy=agentFullName.split(":")
47 if not model.endswith("EscortedAgent"):
48 if policy.startswith("Policy_"):
49 policyName[team]=policy[7:]
50 else:
51 policyName[team]=policy
52 break
53 return policyName
54 def getRewardStringToDisplay(self, team, frame):
55 """報酬を表示する際の文字列を取得するもの。
56 R8年度コンテストでは護衛対象機の報酬を表示対象から除外する。
57 Args:
58 team str: 計算対象の陣営
59 frame dict: 現在のフレームデータ
60
61 Returns:
62 str: 報酬表示用の文字列
63 """
64 def stringifyTotalRewards(tr):
65 if isinstance(tr,float):
66 return "{:.1f}".format(tr)
67 else:
68 if len(tr)==1:
69 return "{:.1f}".format(next(iter(tr.values())))
70 else:
71 return "{"+",".join([k+":{:.1f}".format(v) for k,v in tr.items()])+"}"
72 def check(agentFullName):
73 name,model,policy=agentFullName.split(":")
74 return not model.endswith("EscortedAgent")
75 reward=[stringifyTotalRewards(frame["totalRewards"][agentFullName]) for agentFullName,agent in frame["agents"].items() if agent["team"]==team and check(agentFullName)]
76 return str(reward)
