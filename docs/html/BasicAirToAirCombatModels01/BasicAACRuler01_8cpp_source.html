216 std::string team=asset->getTeam();
217 std::string modelName=asset->getFactoryModelName();

250 hitCount[team]=std::map<std::string,int>();
251 if(manager->getTickCount()==0){

254 for(auto&& e:manager->getAssets([&](const std::shared_ptr<const Asset>& asset)->bool {
255 return asset->isAlive() && asset->getTeam()==team && isinstance<Fighter>(asset) && isToBeConsideredForBreak(team,asset->getFactoryModelName());
256 })){

261 }

264void BasicAACRuler01::onInnerStepEnd(){
265 //生存数の監視
266 std::map<std::string,int> aliveCount;
267 for(auto& team:teams){
268 aliveCount[team]=0;
269 for(auto&& e:manager->getAssets([&](const std::shared_ptr<const Asset>& asset)->bool {
270 return asset->isAlive() && asset->getTeam()==team && isinstance<Fighter>(asset) && isToBeConsideredForElimination(team,asset->getFactoryModelName());
271 })){
272 aliveCount[team]++;
273 }
274 if(aliveCount[team]==0 && eliminatedTime[team]<0){
275 eliminatedTime[team]=manager->getElapsedTime();
276 }
277 }
278 //防衛ラインの突破タイミングを監視
279 std::vector<double> tmp;
280 for(auto& team:teams){
281 tmp.clear();
282 tmp.push_back(-dLine);
283 for(auto&& e:manager->getAssets([&](const std::shared_ptr<const Asset>& asset)->bool {
284 return asset->isAlive() && asset->getTeam()==team && isinstance<Fighter>(asset) && isToBeConsideredForBreak(team,asset->getFactoryModelName());
285 })){
286 auto f=getShared<Fighter>(e);
287 tmp.push_back(forwardAx[team].dot(f->pos(getLocalCRS()).block<2,1>(0,0,2,1)));
288 }
289 leadRange[team]=*std::max_element(tmp.begin(),tmp.end());
290 if(leadRange[team]>=dLine && breakTime[team]<0){
291 breakTime[team]=manager->getElapsedTime();
292 //得点計算 2：突破による加点(pBreak点)
293 debugPrint("2. Break",team,pBreak);
294 stepScore[team]+=pBreak;
295 }
296 }
297
298 for(auto& team:teams){
299 //得点計算 1：撃墜による加点(1機あたりpHit点)
300 for(auto&& c:hitCount[team]){
301 debugPrint("1. Hit("+c.first+")",team,c.second*getPHit(team,c.first));
302 stepScore[team]+=c.second*getPHit(team,c.first);
303 }
304 //得点計算 6-(a)：墜落による減点(1機あたりpCrash点)
305 for(auto&& c:crashCount[team]){
306 debugPrint("6-(a). Crash("+c.first+")",team,-c.second*getPCrash(team,c.first));
307 double penalty = c.second*getPCrash(team,c.first);
308 if(independentPenalty && pCrashAsPenalty){
309 stepPenalty[team]+=penalty;
310 }else{
311 stepScore[team]-=penalty;
312 }
313 }
314 //得点計算 6-(b)：場外に対する減点(1秒、1kmにつきpOut点)
315 outDist[team]=0;
316 for(auto&& e:manager->getAssets([&](const std::shared_ptr<const Asset>& asset)->bool {
317 return asset->isAlive() && asset->getTeam()==team && isinstance<Fighter>(asset);
318 })){
319 auto f=getShared<Fighter>(e);
320 double d=abs(sideAx[team].dot(f->pos(getLocalCRS()).block<2,1>(0,0,2,1)))-dOut;
321 if(regardDefenseLineAsPenaltyLine){
322 d=std::max(d,abs(forwardAx[team].dot(f->pos(getLocalCRS()).block<2,1>(0,0,2,1)))-dLine);
323 }
324 outDist[team]+=std::max(0.0,d);
325 }
326 if(outDist[team]>0.0){
327 debugPrint("6-(b). Out",team,-(outDist[team]/1000.)*pOut*interval[SimPhase::ON_INNERSTEP_END]*manager->getBaseTimeStep());
328 double penalty = (outDist[team]/1000.)*pOut*interval[SimPhase::ON_INNERSTEP_END]*manager->getBaseTimeStep();
329 if(independentPenalty){
330 stepPenalty[team]+=penalty;
331 }else{
332 stepScore[team]-=penalty;
333 }
334 }
335
336 //失格判定
337 if(independentPenalty){
338 if(-(totalPenalty[team]+stepPenalty[team])<=pDisq && disqTime[team]<0){
339 disqTime[team]=manager->getElapsedTime();
340 }
341 }else{
342 if(score[team]+stepScore[team]<=pDisq && disqTime[team]<0){
343 disqTime[team]=manager->getElapsedTime();
344 }
345 }
346 }
347 if(endReasonSub==EndReason::NOTYET){
348 if(terminalAtElimination &&
349 (eliminatedTime[westSider]>=0 || eliminatedTime[eastSider]>=0)
350 ){
351 //終了条件(1)が有効な場合
352 if(!(enableAdditionalTime && checkHitPossibility())){
353 //相討ちを考慮する場合、以後の撃墜の可能性がある限り継続
354 endReasonSub=EndReason::ELIMINATION;
355 }
356 }
357 if(endReasonSub==EndReason::NOTYET){
358 if(terminalAtBreak &&
359 (breakTime[westSider]>=0 || breakTime[eastSider]>=0)
360 ){
361 //終了条件(2)が有効な場合
362 if(!(enableAdditionalTime && checkHitPossibility())){
363 //相討ちを考慮する場合、以後の撃墜の可能性がある限り継続
364 endReasonSub=EndReason::BREAK;
365 }
366 }
367 if(endReasonSub==EndReason::NOTYET){
368 if(disqTime[westSider]>=0 || disqTime[eastSider]>=0){
369 //終了条件(5)の判定
370 endReasonSub=EndReason::PENALTY;
371 }else{
372 //終了条件(3)の判定
373 bool withdrawal=manager->getElapsedTime()>=minTime;
374 for(auto& team:teams){
375 for(auto&& e:manager->getAssets([&](const std::shared_ptr<const Asset>& asset)->bool {
376 return asset->isAlive() && asset->getTeam()==team && isinstance<Fighter>(asset);
377 })){
378 auto f=getShared<Fighter>(e);
379 if(forwardAx[team].dot(f->pos(getLocalCRS()).block<2,1>(0,0,2,1))>=-dLine){
380 withdrawal=false;
381 break;
382 }
383 }
384 if(!withdrawal){
385 break;
386 }
387 }
388 if(withdrawal){
389 endReasonSub=EndReason::WITHDRAWAL;
390 }
391 }
392 }
393 }
394 }

396void BasicAACRuler01::checkDone(){
397 //終了判定
398 dones.clear();
399 for(auto&& e:manager->getAgents()){
400 auto a=getShared(e);
401 dones[a->getFullName()]=!a->isAlive();
402 }
403 bool considerAdvantage=false;
404 //終了条件(1)：全機撃墜or墜落
405 if(endReasonSub==EndReason::ELIMINATION){
406 endReason=EndReason::ELIMINATION;
407 if(breakTime[westSider]<0 && breakTime[eastSider]<0){
408 //両者未突破の場合、進出度合いによる優勢度と残存陣営による突破可能性の考慮が必要
409 if(eliminatedTime[westSider]>=0 && eliminatedTime[eastSider]>=0){
410 //両者全滅した場合、進出度合いによる優勢度を考慮
411 considerAdvantage=true;
412 for(auto&& t:teams){
413 leadRange[t]=lastDownPosition[t];
414 }
415 }else{
416 //一方が生存していた場合
417 //未突破の陣営に「そこから突破して更に帰還可能な」突破判定対象機が存在していれば+pBreak点
418 if(eliminatedTime[eastSider]<0){//東が生存
419 bool chk=false;
420 for(auto&& e:manager->getAssets([&](const std::shared_ptr<const Asset>& asset)->bool {
421 return asset->isAlive() && asset->getTeam()==eastSider && isinstance<Fighter>(asset) && isToBeConsideredForBreak(eastSider,asset->getFactoryModelName());
422 })){
423 if(isBreakableAndReturnableToBase(e.lock())){
424 chk=true;
425 break;
426 }
427 }
428 if(chk){
429 debugPrint("3. Break",eastSider,pBreak);
430 stepScore[eastSider]+=pBreak;
431 score[eastSider]+=pBreak;
432 }
433 }else{//西が生存
434 bool chk=false;
435 for(auto&& e:manager->getAssets([&](const std::shared_ptr<const Asset>& asset)->bool {
436 return asset->isAlive() && asset->getTeam()==westSider && isinstance<Fighter>(asset) && isToBeConsideredForBreak(westSider,asset->getFactoryModelName());
437 })){
438 if(isBreakableAndReturnableToBase(e.lock())){
439 chk=true;
440 break;
441 }
442 }
443 if(chk){
444 debugPrint("3. Break",westSider,pBreak);
445 stepScore[westSider]+=pBreak;
446 score[westSider]+=pBreak;
447 }
448 }
449 }
450 }
451 }
452 //終了条件(2)：防衛ラインの突破
453 else if(endReasonSub==EndReason::BREAK){
454 endReason=EndReason::BREAK;
455 //追加の得点増減はなし
456 }
457 //終了条件(5)：ペナルティによる敗北
458 else if(endReasonSub==EndReason::PENALTY){
459 endReason=EndReason::PENALTY;
460 if(breakTime[westSider]<0 && breakTime[eastSider]<0){
461 //両者未突破の場合、進出度合いによる優勢度を計算
462 considerAdvantage=true;
463 }
464 }
465 //終了条件(3)：両者撤退による打ち切り
466 else if(endReasonSub==EndReason::WITHDRAWAL){
467 endReason=EndReason::WITHDRAWAL;
468 //追加の得点増減はなし(優勢度も計算しない)
469 }
470 //終了条件(4)：時間切れ
471 else if(maxTime>0 && manager->getElapsedTime()>=maxTime){
472 endReason=EndReason::TIMEUP;
473 if(breakTime[westSider]<0 && breakTime[eastSider]<0){
474 //両者未突破の場合、得点計算5に従い優勢度を計算
475 considerAdvantage=true;
476 }
477 }
478
479 //生存点
480 //帰還不可による墜落ペナルティ
481 if(endReason!=EndReason::NOTYET){
482 for(auto& team:teams){
483 for(auto&& e:manager->getAssets([&](const std::shared_ptr<const Asset>& asset)->bool {
484 return asset->isAlive() && asset->getTeam()==team && isinstance<Fighter>(asset) && isToBeConsideredForElimination(team,asset->getFactoryModelName());
485 })){
486 auto asset=e.lock();
487 if(isReturnableToBase(asset)){
488 //帰還可能なら生存点
489 debugPrint("4. Alive("+asset->getFactoryModelName()+")",team,getPAlive(team,asset->getFactoryModelName()));
490 stepScore[team]+=getPAlive(team,asset->getFactoryModelName());
491 score[team]+=getPAlive(team,asset->getFactoryModelName());
492 }else{
493 //帰還不可能なら墜落ペナルティ
494 double penalty = getPCrash(team,asset->getFactoryModelName());
495 debugPrint("6(a). NoReturn("+asset->getFactoryModelName()+")",team,-penalty);
496 if(independentPenalty && pCrashAsPenalty){
497 stepPenalty[team]+=penalty;
498 totalPenalty[team]+=penalty;
499 }else{
500 stepScore[team]-=penalty;
501 score[team]-=penalty;
502 }
503 }
504 }
505 }
506 }
507
508 if(considerAdvantage){
509 //終了条件(4)、(5)または両者全滅の場合で、両者未突破だった場合の優勢度加点
510 //得点計算 5：進出度合いに対する加点(1kmにつきpAdv点)
511 if(leadRange[westSider]>leadRange[eastSider]){
512 double s=(leadRange[westSider]-leadRange[eastSider])/2./1000.*pAdv;
513 debugPrint("5. Adv",westSider,s);
514 stepScore[westSider]+=s;
515 score[westSider]+=s;
516 }else{
517 double s=(leadRange[eastSider]-leadRange[westSider])/2./1000.*pAdv;
518 debugPrint("5. Adv",eastSider,s);
519 stepScore[eastSider]+=s;
520 score[eastSider]+=s;
521 }
522 }
523 //打ち切り対策で常時勝者を仮設定しておく
524 if(independentPenalty && endReason==EndReason::PENALTY){
525 // independentPenaltyがtrueの場合は失格時の勝敗判定に得点を用いない。
526 if(disqTime[westSider]>=0 && disqTime[eastSider]>=0){
527 // 両方失格の場合は引き分け
528 winner="";
529 }else if(disqTime[westSider]>=0){
530 // 西が失格の場合
531 winner=eastSider;
532 }else{
533 // 東が失格の場合
534 winner=westSider;
535 }
536 }else{
537 if(score[westSider]>score[eastSider]){
538 winner=westSider;
539 }else if(score[westSider]<score[eastSider]){
540 winner=eastSider;
541 }else{
542 winner="";//引き分けは空文字列
543 }
544 }
545 if(endReason!=EndReason::NOTYET){
546 for(auto& e:dones){
547 e.second=true;
548 }
549 dones["__all__"]=true;
550 }else{
551 dones["__all__"]=false;
552 }
