113bool Asset::isAlive() const{
114 if(isComponent(isManaged())){
115 // 他のAssetのcomponentのとき、holderがaliveでなければ自身も連動して非alive
116 if(isHolderExpired(isManaged()) || !holder<Asset>(isManaged())->isAlive()){
117 return false;
118 }
119 }
120 // 他のAssetによって決まらない場合は、自身固有のフラグで判定する。
121 return _isAlive;
122}

124bool Asset::removeWhenKilled() const{

126 // 他のAssetのcomponentのとき、holderがkill時にremoveされるなら連動してremove

128 return true;

130 }
131 // 他のAssetによって決まらない場合は、自身固有のフラグで判定する。
