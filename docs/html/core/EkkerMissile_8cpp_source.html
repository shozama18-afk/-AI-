80void EkkerMissile::calcMotion(double tAftLaunch,double dt){
81 motion_prev=motion;
82 double V=motion.vel.norm();
83 auto propNav=getShared<PropNav>(controllers["Navigator"]);
84 Eigen::Vector3d accelCmd=propNav->commands["accel"];
85 Eigen::Vector3d omegaCmd=propNav->commands["omega"];
86 double omega=omegaCmd.norm();
87 double omegaMax=maxLoadG*gravity/V;
88 if(omega>omegaMax){
89 omegaCmd*=(omegaMax/omega);
90 omega=omegaMax;
91 accelCmd=omegaCmd.cross(motion.vel());
92 }
93 auto [accelScalar,omegaV] = calcAccel(accelCmd,tAftLaunch,dt);
94 Eigen::Vector3d ex=motion.vel()/V;
95 Eigen::Vector3d accel=ex*accelScalar;
96 omega=omegaV.norm();
97 if(omega*dt<1e-8){//ほぼ無回転
98 motion.setPos(motion.pos()+motion.vel()*dt+accel*dt*dt/2.);
99 motion.setVel(motion.vel()+accel*dt);
100 }else{
101 Eigen::Vector3d ex0=motion.vel()/V;
102 Eigen::Vector3d ey0=omegaV.cross(ex0).normalized();
103 double wt=omega*dt;
104 double acc=accelScalar;
105 Eigen::Vector3d vn=(ex0*cos(wt)+ey0*sin(wt)).normalized();
106 motion.setVel((V+acc*dt)*vn);
107 Eigen::Vector3d dr=V*dt*(ex0*sinc(wt)+ey0*oneMinusCos_x2(wt)*wt)+acc*dt*dt*(ex0*(sinc(wt)-oneMinusCos_x2(wt))+ey0*wt*(sincMinusOne_x2(wt)+oneMinusCos_x2(wt)));
108 motion.setPos(motion.pos()+dr);
109 accel=(motion.vel()-vel_prev())/dt;
110 }
111 motion.setOmega(omegaV);
112 calcQ();
113}

115std::pair<double,Eigen::Vector3d> EkkerMissile::calcAccel(const Eigen::Vector3d& accelCmd, double tAftLaunch,double dt){
116 double V=motion.vel.norm();
117 double alt=motion.getHeight();
118 bool pwroff=tAftLaunch>=tBurn;
119 double thrust_;
120 if(pwroff){
121 thrust_ = 0.0;
122 mass = massF;
123 lcg = lcgf;
124 }else{
125 thrust_ = thrust;
126 mass = massI+(massF-massI)*tAftLaunch/tBurn;
127 lcg = lcgi+(lcgf-lcgi)*tAftLaunch/tBurn;
128 }
129 std::map<std::string,double> atm=atmosphere(alt);
130 double M=V/atm["a"];
131 double qd=0.5*atm["rho"]*V*V;
132 Eigen::Vector3d ex=motion.vel()/V;
133 Eigen::Vector3d gravityAccelVector=motion.dirHtoP(Eigen::Vector3d(0,0,1),"NED")*gravity; //TODO 慣性力モデルに置換
134 Eigen::Vector3d ey=(accelCmd-gravityAccelVector).cross(ex)*mass;
135 double desiredSideForce=ey.norm();
136 Eigen::Vector3d sideAx;
137 double aoa,CL,CD;
138 if(desiredSideForce>0){
139 ey/=desiredSideForce;
140 sideAx=ex.cross(ey);
141 std::pair<double,double> tmp=EkkerMDT::cldcmd(d,l,ln,lcg,bt,St,M);
142 double cld=tmp.first;
143 double cmd=tmp.second;
144 auto func=[&](double aoa_){
145 std::pair<double,double> tmp=EkkerMDT::clacma(d,l,ln,lcg,bw,Sw,lw,bt,St,M,aoa_);
146 double cla=tmp.first;
147 double cma=tmp.second;
148 double delta=std::clamp<double>(-cma/cmd*aoa_,-maxD,maxD);
149 return (cla*aoa_+cld*delta)*(qd*Sref)+thrust_*sin(aoa_)-desiredSideForce;
150 };
151 if(func(0.0)>=0){
152 aoa=0.0;
153 }else if(func(maxA)<=0){
154 aoa=maxA;
155 }else{
156 try{
157 boost::math::tools::eps_tolerance<double> tol(16);
158 boost::uintmax_t max_iter=10;
159 auto result=boost::math::tools::toms748_solve(func,0.0,maxA,tol,max_iter);
160 aoa=(result.first+result.second)/2;
161 }catch(std::exception& e){
162 std::cout<<"exception at EkkerMissile::calcAccel"<<std::endl;
163 DEBUG_PRINT_EXPRESSION(func(0.0))
164 DEBUG_PRINT_EXPRESSION(func(maxA))
165 DEBUG_PRINT_EXPRESSION(alt)
166 DEBUG_PRINT_EXPRESSION(M)
167 DEBUG_PRINT_EXPRESSION(desiredSideForce)
168 DEBUG_PRINT_EXPRESSION(thrust_)
169 throw e;
170 }
171 }
172 tmp=EkkerMDT::clacma(d,l,ln,lcg,bw,Sw,lw,bt,St,M,aoa);
173 double cla=tmp.first;
174 double cma=tmp.second;
175 double delta=std::clamp<double>(-cma/cmd*aoa,-maxD,maxD);
176 //CL=(cla-cld*cma/cmd)*aoa;
177 CL=cla*aoa+cld*delta;
178 CD=EkkerMDT::cdtrim(d,l,ln,bw,Sw,thicknessRatio,bt,St,thicknessRatio,M,alt,aoa,delta,frictionFactor,pwroff);
179 }else{
180 sideAx<<0,0,0;
181 CL=0.0;
182 CD=EkkerMDT::cdtrim(d,l,ln,bw,Sw,thicknessRatio,bt,St,thicknessRatio,M,alt,0,0,frictionFactor,pwroff);
183 aoa=0.0;
184 }
185 Eigen::Vector3d sideAccel=sideAx*(CL*qd*Sref+thrust_*sin(aoa))/mass+(gravityAccelVector-ex*(gravityAccelVector.dot(ex)));
186 //attitude and deflection are assumed to be immidiately adjustable as desired.
187 double accelScalar=(-CD*Sref*qd+thrust_*cos(aoa))/mass+gravityAccelVector.dot(ex);
188 Eigen::Vector3d omegaV=motion.vel().cross(sideAccel)/(V*V);
189 return {accelScalar,omegaV};
190}
191bool EkkerMissile::hitCheck(const Eigen::Vector3d &tpos,const Eigen::Vector3d &tpos_prev){
192 Eigen::Vector3d r1=pos_prev();
193 Eigen::Vector3d r2=tpos_prev;
194 Eigen::Vector3d d1=pos()-r1;
195 Eigen::Vector3d d2=tpos-r2;
196 Eigen::Vector3d A=r1-r2;double a=A.norm();
197 Eigen::Vector3d B=d1-d2;double b=B.norm();
198 if(a<hitD){//初期位置であたり
199 return true;
200 }else if(b<1e-8){//相対速度がほぼ0}
201 return a<hitD;
202 }
203 double tMin=std::min(1.,std::max(0.,-A.dot(B)/(b*b)));
204 return (A+B*tMin).norm()<hitD;
205}
