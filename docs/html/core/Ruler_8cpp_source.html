65void Ruler::onEpisodeBegin(){
66 teams.clear();
67 score.clear();
68 stepScore.clear();
69 stepPenalty.clear();
70 totalPenalty.clear();
71 nl::json j_teams;
72 try{
73 j_teams=modelConfig.at("teams");
74 }catch(...){
75 j_teams="All";
76 }
77 if(j_teams.is_null()){
78 j_teams="All";
79 }
80 if(j_teams.is_string()){
81 std::string team=j_teams;
82 if(team=="All"){
83 for(auto &&t:manager->getTeams()){
84 teams.push_back(t);
85 score[t]=0.0;
86 stepScore[t]=0.0;
87 stepPenalty[t]=0.0;
88 totalPenalty[t]=0.0;
89 }
90 }else{
91 teams.push_back(team);
92 score[team]=0.0;
93 stepScore[team]=0.0;
94 stepPenalty[team]=0.0;
95 totalPenalty[team]=0.0;
96 }
97 }else if(j_teams.is_array()){
98 teams=j_teams.get<std::vector<std::string>>();
99 for(auto &&t:teams){
100 stepPenalty[t]=0.0;
101 totalPenalty[t]=0.0;
102 }
103 }else{
104 std::cout<<"modelConfig['teams']="<<j_teams<<std::endl;
105 throw std::runtime_error("Invalid designation of 'teams' for Ruler.");
106 }
107 dones.clear();
108 for(auto &&e:manager->getAgents()){
109 auto a=getShared(e);
110 dones[a->getName()]=!a->isAlive();
111 }
112 dones["__all__"]=false;
113 endReason=EndReason::NOTYET;
114 observables["endReason"]=endReason;

135void Ruler::checkDone(){
136 dones.clear();
137 for(auto &&e:manager->getAgents()){
138 auto a=getShared(e);
139 dones[a->getName()]=!a->isAlive();
140 }
141 if(maxTime>0 && manager->getElapsedTime()>=maxTime){
142 int index=0;
143 double maxScore=0;
144 for(auto &&t:teams){
145 if(index==0 || maxScore<score[t]){
146 maxScore=score[t];
147 winner=t;
148 }
149 index++;
150 }
151 for(auto &e:dones){
152 e.second=true;
153 }
154 dones["__all__"]=true;
155 endReason=EndReason::TIMEUP;
156 return;
157 }
158 if(dones.size()==0){
159 dones["__all__"]=false;
160 return;
161 }
162 dones["__all__"]=true;
163 for(auto &&e:dones){
164 if(!e.second){
165 dones["__all__"]=false;
166 }
167 }
168 if(dones["__all__"]){
169 endReason=EndReason::NO_ONE_ALIVE;
170 }
171 observables["endReason"]=endReason;
172}
