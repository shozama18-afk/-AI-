92std::pair<Eigen::Vector3d,double> CapsuleCoursePatrolAgent::getCoursePoint(const double& phase) const{
93 Eigen::Vector3d dp=p2-p1;
94 double distance=sqrt(dp(0)*dp(0)+dp(1)*dp(1));
95 Eigen::Vector3d ex=dp.normalized();
96 Eigen::Vector3d ey=Eigen::Vector3d(0,0,1).cross(ex).normalized();
97 double phase2rad=course_length/radius;
98
99 if(isClockwise != "right"){
100 ey = Eigen::Vector3d(0,0,-1).cross(ex).normalized();
101 }
102
103 std::vector<double> anchors = {
104 (radius*M_PI_2)/course_length,
105 (radius*M_PI_2+distance)/course_length,
106 (radius*M_PI_2*3+distance)/course_length,
107 (radius*M_PI_2*3+2*distance)/course_length
108 };
109 if(phase<anchors[0]){
110 double angle=phase*phase2rad;
111 return {
112 p1+radius*(-ex*cos(angle)-ey*sin(angle)),
113 angle
114 };
115 }else if(phase<anchors[1]){
116 return {
117 p1+(ex*(phase-anchors[0])*course_length-ey*radius),
118 M_PI
119 };
120 }else if(phase<anchors[2]){
121 double angle=(phase-anchors[1])*phase2rad;
122 return {
123 p2+radius*(ex*sin(angle)-ey*cos(angle)),
124 M_PI_2+angle
125 };
126 }else if(phase<anchors[3]){
127 return {
128 p2+(-ex*(phase-anchors[2])*course_length+ey*radius),
129 M_PI
130 };
131 }else{
132 double angle=(1.0-phase)*phase2rad;
133 return {
134 p1+radius*(-ex*cos(angle)+ey*sin(angle)),
135 -angle
136 };
137 }
138}

140double CapsuleCoursePatrolAgent::getCoursePhase(const Eigen::Vector3d& pos) const{
141 Eigen::Vector3d dp=p2-p1;
142 double distance=sqrt(dp(0)*dp(0)+dp(1)*dp(1));
143 Eigen::Vector3d ex=dp.normalized();
144 Eigen::Vector3d ey=Eigen::Vector3d(0,0,1).cross(ex).normalized();
145 double rad2phase=radius/course_length;
146
147 if(isClockwise != "right"){
148 ey = Eigen::Vector3d(0,0,-1).cross(ex).normalized();
149 }
150
151
152 std::vector<double> anchors = {
153 (radius*M_PI_2)/course_length,
154 (radius*M_PI_2+distance)/course_length,
155 (radius*M_PI_2*3+distance)/course_length,
156 (radius*M_PI_2*3+2*distance)/course_length
157 };
158
159 double x=ex.dot(pos-p1);
160 double y=ey.dot(pos-p1);
161 if(x<0){
162 double angle=atan2(-y,-x);
163 return angle*rad2phase;
164 }else if(x>distance){
165 double angle=atan2(x-distance,-y);
166 return anchors[1]+angle*rad2phase;
167 }else{
168 if(y<0){
169 return anchors[0]+x/course_length;
170 }else{
171 return anchors[2]+(distance-x)/course_length;
172 }
173 }

177void CapsuleCoursePatrolAgent::control(){
178 MotionState originalMyMotion(parent->observables.at("motion"));
179 MotionState myMotion=originalMyMotion.transformTo(getLocalCRS());
180 Eigen::Vector3d pos=myMotion.pos();
181 Eigen::Vector3d vel=myMotion.vel();
182
183 velocity=getValueFromJsonKRD(modelConfig,"velocity",randomGen,250.0);
184
185 Eigen::Vector3d dstDir;
186 double dstAlt;
187 if(chkMWS()){
188 std::vector<std::pair<double,Eigen::Vector3d>> tmp;
189 for(auto& m:parent->observables.at("/sensor/mws/track"_json_pointer)){
190 Eigen::Vector3d dir=m.get<Track2D>().dir(getLocalCRS());
191 tmp.push_back(std::make_pair(-dir.dot(myMotion.dirBtoP(Eigen::Vector3d(1,0,0))),-dir));
192 }
193 Eigen::Vector3d dr=(*std::min_element(tmp.begin(),tmp.end(),
194 [](const std::pair<double,Eigen::Vector3d> &lhs,const std::pair<double,Eigen::Vector3d> &rhs){
195 return lhs.first<rhs.first;
196 })).second;
197 dstDir=dr;
198 dstDir(2)=0;
199 dstDir.normalize();
200 double dAz=atan2(dstDir(1),dstDir(0))-myMotion.getAZ();
201 while(dAz>M_PI){dAz-=2*M_PI;}
202 while(dAz<=-M_PI){dAz+=2*M_PI;}
203 if(dAz > deltaAZLimit){
204 dstDir << cos(myMotion.getAZ()+deltaAZLimit), sin(myMotion.getAZ()+deltaAZLimit), 0;
205 }else if(dAz < -deltaAZLimit){
206 dstDir << cos(myMotion.getAZ()-deltaAZLimit), sin(myMotion.getAZ()-deltaAZLimit), 0;
207 }
208 altitudeKeeper.minPitch=minPitchForEvasion;
209 altitudeKeeper.maxPitch=maxPitchForEvasion;
210 dstDir=altitudeKeeper(myMotion,dstDir,minAltForEvasion);
211 dstAlt=minAltForEvasion;
212 }else{
213 double currentPhase=getCoursePhase(pos);
214 double dt=interval[SimPhase::CONTROL]*manager->getBaseTimeStep()*10;
215 double delta = velocity*dt/course_length;
216 auto [targetPoint, targetHeading] = getCoursePoint(currentPhase+delta);
217 dstDir=targetPoint-pos;
218 dstDir(2)=0;
219 dstDir.normalize();
220 double dAz=atan2(dstDir(1),dstDir(0))-myMotion.getAZ();
221 while(dAz>M_PI){dAz-=2*M_PI;}
222 while(dAz<=-M_PI){dAz+=2*M_PI;}
223 if(dAz > deltaAZLimit){
224 dstDir << cos(myMotion.getAZ()+deltaAZLimit), sin(myMotion.getAZ()+deltaAZLimit), 0;
225 }else if(dAz < -deltaAZLimit){
226 dstDir << cos(myMotion.getAZ()-deltaAZLimit), sin(myMotion.getAZ()-deltaAZLimit), 0;
227 }
228 altitudeKeeper.minPitch=minPitchForNormal;
229 altitudeKeeper.maxPitch=maxPitchForNormal;
230 dstDir=altitudeKeeper(myMotion,dstDir,-targetPoint(2));
231 dstAlt=-targetPoint(2);
232 }
233
234 observables[parent->getFullName()]["decision"]={
235 {"Roll",nl::json::array({"Don't care"})},
236 {"Horizontal",nl::json::array({"Az_NED",atan2(dstDir(1),dstDir(0))})},
237 {"Vertical",nl::json::array({"Alt",dstAlt})},
238 {"Throttle",nl::json::array({"Vel",velocity})},
239 {"Fire",nl::json::array({false,Track3D()})}
240 };
241
242 commands[parent->getFullName()]={
243 {"motion",{
244 {"dstDir",originalMyMotion.dirAtoP(dstDir,myMotion.pos(),getLocalCRS())}//元のparent座標系に戻す
245 }},
246 {"weapon",{
247 {"launch",false},
248 {"target",Track3D()}
249 }}
250 };
251 commands[parent->getFullName()]["motion"]["dstV"]=velocity;
252}
