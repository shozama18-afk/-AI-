310void BasicAACRuleBasedAgent01::updateTrackInfo(TrackInfo& ti,bool isFull){
311 /*TrackInfoの付帯情報の更新を行う。
312 * 射程情報: 双方の距離,RmaxHead,Rmax1,Rmax2
313 * 追尾状況: 自機が追尾中か否か,誰かが追尾中か否か,追尾機数(IN,LIMIT,OUT) ただし、stateがEVADEまたはWITHDRAWの機体はINでもLIMIT扱いとする。
314 * isFullにTrueを入れたときはdeploy用に射程計算等を含む全てを計算し、Falseを入れたときは位置と速度だけ更新する。
315 */
316 Eigen::Vector3d tpos=ti.pos(getLocalCRS());
317 Eigen::Vector3d tvel=ti.vel(getLocalCRS());
318 Eigen::Vector3d dpos=pos-tpos;
319 double L=dpos.norm();
320 ti.distance=L;
321 if(isFull && ti.state!=TrackInfo::UpdateState::LOST){
322 ti.myRHead=calcRHead(pos,vel,tpos,tvel);
323 ti.myRTail=calcRTail(pos,vel,tpos,tvel);
324 ti.hisRHead=calcRHead(tpos,tvel,pos,vel);
325 ti.hisRTail=calcRTail(tpos,tvel,pos,vel);
326 Eigen::Vector3d tEx=tvel.normalized();
327 if(ti.state==TrackInfo::UpdateState::TRACK){
328 ti.trackers.clear();
329 for(auto& src:parent->observables.at("/sensor/trackSource"_json_pointer)[ti.idx]){
330 if(std::find(ti.trackers.begin(),ti.trackers.end(),src.get<std::string>())==ti.trackers.end()){
331 ti.trackers.push_back(src);
332 }
333 }
334 ti.numTracker=ti.trackers.size();
335 ti.nonLimitTrackers.clear();
336 ti.limitTrackers.clear();
337 ti.numTrackerLimit=0;
338 ti.inOurSensor=TrackInfo::SensingState::OUTSIDE;
339 ti.inMySensor=TrackInfo::SensingState::OUTSIDE;
340 for(auto& e:ti.trackers){
341 const nl::json& fObs=e==parent->getFullName() ?
342 parent->observables :
343 parent->observables.at("/shared/fighter"_json_pointer).at(e);
344 const nl::json& aObs=parent->observables.at("/shared/agent"_json_pointer).at(e);
345 State aState=State::NONE;
346 try{
347 aObs.at("state").get_to(aState);
348 }catch(...){
349 }
350 Eigen::Vector3d tposS=MotionState(fObs.at("motion")).absAtoB(tpos,getLocalCRS());
351 double L=tposS.norm();
352 double angle=acos(std::max(-1.0,std::min(1.0,tposS(0)/L)));
353 double fLref,fAngle;
354 if(fObs.contains("/spec/sensor/radar"_json_pointer)){
355 fLref=fObs.at("/spec/sensor/radar/Lref"_json_pointer);
356 fAngle=fObs.at("/spec/sensor/radar/thetaFOR"_json_pointer);
357 }else{
358 fLref=0;
359 fAngle=0;
360 }
361 if(
362 L<=fLref-sensorInRangeLimit &&
363 angle<=fAngle-sensorInCoverageLimit &&
364 (!considerStateForTrackingCondition || (
365 aState!=State::EVADE &&
366 aState!=State::WITHDRAW &&
367 aState!=State::NONE
368 ))
369 ){
370 ti.inOurSensor=TrackInfo::SensingState::INSIDE;
371 ti.nonLimitTrackers.push_back(e);
372 if(e==parent->getFullName()){
373 ti.inMySensor=TrackInfo::SensingState::INSIDE;
374 }
375 }else if(L<=fLref && angle<=fAngle){
376 ti.limitTrackers.push_back(e);
377 ti.numTrackerLimit++;
378 if(ti.inOurSensor!=TrackInfo::SensingState::INSIDE){
379 ti.inOurSensor=TrackInfo::SensingState::LIMIT;
380 }
381 if(e==parent->getFullName()){
382 if(ti.inMySensor!=TrackInfo::SensingState::INSIDE){
383 ti.inMySensor=TrackInfo::SensingState::LIMIT;
384 }
385 }
386 }
387 }
388 }else{
389 ti.trackers.clear();
390 ti.nonLimitTrackers.clear();
391 ti.limitTrackers.clear();
392 ti.numTracker=0;
393 ti.numTrackerLimit=0;
394 ti.inOurSensor=TrackInfo::SensingState::OUTSIDE;
395 ti.inMySensor=TrackInfo::SensingState::OUTSIDE;
396 }
397 }

399void BasicAACRuleBasedAgent01::updateTargetStatus(bool isFull){
400 /*主目標と副目標を更新する。副目標は、前回の各機の主目標と、まだ誘導が必要な全ての誘導弾目標とし、副目標内でのtruthの重複はないようにする。
401 「誘導が必要」とは、誘導弾自身のセンサで捉えていない状態(mode!=SELF)を指すものとする。
402 なお、副目標には自機の主目標も含む(敢えて除外していない)
403 isFullにTrueを入れたときは副目標の再抽出も行い全て更新するが、Falseを入れたときはその時点で抽出されているものの位置と速度だけ更新する。
404 */
405 //主目標の更新
406 updateTargetStatusSub(target,isFull);
407 updateTargetStatusSub(launchedTarget,isFull);
408 //副目標の確定(isFullの時のみ)
409 if(isFull){
410 std::vector<bool> flag;//いらないものを削除するためのフラグ
411 for(auto& t:additionalTargets){
412 flag.push_back(false);
413 }
414 for(auto&& fr:parent->observables.at("/shared/fighter"_json_pointer).items()){
415 if(fr.value().at("isAlive")){
416 const nl::json& ag=parent->observables.at("/shared/agent"_json_pointer).at(fr.key());
417 TrackInfo tgt;
418 //主目標
419 if(ag.contains("target")){
420 tgt=TrackInfo(ag.at("target"),-1);
421 if(!tgt.is_none()){
422 bool ret=true;
423 int idx=0;
424 for(auto& aTgt:additionalTargets){
425 if(aTgt.isSame(tgt)){
426 ret=false;
427 flag[idx]=true;
428 break;
429 }
430 idx++;
431 }
432 if(ret){
433 additionalTargets.push_back(tgt);
434 flag.push_back(true);
435 }
436 }
437 }
438 //直近の射撃目標
439 if(ag.contains("launchedTarget")){
440 tgt=TrackInfo(ag.at("launchedTarget"),-1);
441 if(!tgt.is_none()){
442 int nextMsl=fr.value().at("/weapon/nextMsl"_json_pointer);
443 if(nextMsl>0){
444 const nl::json& msl=fr.value().at("/weapon/missiles"_json_pointer)[nextMsl-1];
445 if(msl.at("isAlive").get<bool>() && msl.at("mode").get<Missile::Mode>()!=Missile::Mode::SELF){//まだ誘導が必要
446 bool ret=true;
447 int idx=0;
448 for(auto& aTgt:additionalTargets){
449 if(aTgt.isSame(tgt)){
450 ret=false;
451 flag[idx]=true;
452 break;
453 }
454 idx++;
455 }
456 if(ret){
457 additionalTargets.push_back(tgt);
458 flag.push_back(true);
459 }
460 }
461 }
462 }
463 }
464 }
465 }
466 //まだ誘導が必要な誘導弾の目標は直近のでなくても残す
467 for(auto&& msl:parent->observables.at("/weapon/missiles"_json_pointer)){
468 if(msl.at("isAlive").get<bool>() && msl.at("hasLaunched").get<bool>() && msl.at("mode").get<Missile::Mode>()!=Missile::Mode::SELF){
469 int idx=0;
470 for(auto& aTgt:additionalTargets){
471 if(msl.at("target").get<Track3D>().isSame(aTgt)){
472 flag[idx]=true;
473 idx++;
474 }
475 }
476 }
477 }
478 for(auto&& fr:parent->observables.at("/shared/fighter"_json_pointer).items()){
479 if(fr.key()!=parent->getFullName()){
480 for(auto&& msl:fr.value().at("/weapon/missiles"_json_pointer)){
481 if(msl.at("isAlive").get<bool>() && msl.at("hasLaunched").get<bool>() && msl.at("mode").get<Missile::Mode>()!=Missile::Mode::SELF){
482 int idx=0;
483 for(auto& aTgt:additionalTargets){
484 if(msl.at("target").get<Track3D>().isSame(aTgt)){
485 flag[idx]=true;
486 idx++;
487 }
488 }
489 }
490 }
491 }
492 }
493 auto ait=additionalTargets.begin();
494 for(auto fit=flag.begin();fit!=flag.end();){
495 if((*fit) && (*ait).state!=TrackInfo::UpdateState::LOST){
496 ++fit;
497 ++ait;
498 }else{
499 fit=flag.erase(fit);
500 ait=additionalTargets.erase(ait);
501 }
502 }
503 }
504 //副目標の更新
505 for(auto& aTgt:additionalTargets){
506 updateTargetStatusSub(aTgt,isFull);
507 }
508 //行動判断の考慮対象(航跡＋メモリトラック中の主or副目標)の抽出
509 allEnemies.clear();
510 for(auto& t:track){
511 allEnemies.push_back(t);
512 }
513 for(auto& t:additionalTargets){
514 if(t.state==TrackInfo::UpdateState::MEMORY){
515 allEnemies.push_back(t);
516 }
517 }

519void BasicAACRuleBasedAgent01::updateTargetStatusSub(TrackInfo& tgt,bool isFull){
520 /*指定した目標情報を更新する。
521 */
522 if(!tgt.is_none()){
523 bool updated=false;
524 for(auto& t:track){//航跡情報から更新対象を検索
525 if(tgt.isSame(t)){
526 tgt.update(t,isFull);
527 updated=true;
528 break;
529 }
530 }
531 if(!updated){//見つからなかった場合、メモリトラック
532 if(tgt.state==TrackInfo::UpdateState::TRACK){
533 tgt.memoryStartTime=manager->getTime();
534 tgt.state=TrackInfo::UpdateState::MEMORY;
535 }else if(tgt.state==TrackInfo::UpdateState::MEMORY){
536 if(manager->getTime()>=tMaxMemory+tgt.memoryStartTime){
537 tgt.state=TrackInfo::UpdateState::LOST;
538 }
539 }
540 }
541 if(tgt.state!=TrackInfo::UpdateState::TRACK){//外挿
542 tgt.extrapolateTo(manager->getTime());
543 updateTrackInfo(tgt,isFull);
544 }
545 }

576std::pair<bool,std::pair<TrackInfo,double>> BasicAACRuleBasedAgent01::chkConditionForShoot(TrackInfo& tgt){
577 /*(c1)射撃条件の判定
578 1. 射程による判定・・・その航跡に対してRtailを0、Rheadを1としたときに現在の距離がkShoot以下であること
579 2. 探知状況による判定・・・その航跡を自身か味方の少なくとも１機が余裕をもって捉えていること
580 3. 射撃状況による判定・・・その航跡に対して自身が発射した飛翔中の誘導弾がnMslSimul発未満であること
581 */
582 //条件0：機体側の制限も確認しておく
583 bool ret0=getShared<FighterAccessor>(parent)->isLaunchableAt(tgt);
584 //条件1：射程による判定
585 bool ret1=!tgt.is_none();
586 double r;
587 if(ret1){
588 r=tgt.distance-tgt.myRTail;
589 double delta=tgt.myRHead-tgt.myRTail;
590 if(delta==0){
591 if(r<0){
592 r=-std::numeric_limits<double>::infinity();
593 }else if(r>0){
594 r=std::numeric_limits<double>::infinity();
595 }else{
596 r=0;
597 }
598 }else{
599 r/=delta;
600 }
601 ret1=r<=kShoot;
602 }else{
603 r=0;
604 }
605 //条件2：探知状況による判定
606 bool ret2=false;
607 for(auto& f:tgt.nonLimitTrackers){
608 State aState=State::NONE;
609 try{
610 parent->observables.at("/shared/agent"_json_pointer).at(f).at("state").get_to(aState);
611 }catch(...){
612 }
613 if(!considerStateForTrackingCondition || (aState!=State::EVADE && aState!=State::WITHDRAW&& aState!=State::NONE)){
614 ret2=true;
615 break;
616 }
617 }
618 //条件3:射撃状況による判定
619 bool ret3=true;
620 int count=0;
621 for(auto&& msl:parent->observables.at("/weapon/missiles"_json_pointer)){
622 if(msl.at("isAlive").get<bool>() && msl.at("hasLaunched").get<bool>()){
623 if(msl.at("target").get<Track3D>().isSame(tgt)){
624 count++;
625 if(count>=nMslSimul){
626 ret3=false;
627 break;
628 }
629 }
630 }
631 }
632 return std::make_pair(ret0 && ret1 && ret2 && ret3,std::make_pair(tgt,r));
633}

680void BasicAACRuleBasedAgent01::selectTarget(){
681 /*主目標の選択を行う。候補は全航跡に加え、メモリトラック中の副目標(主目標も含む)とする
682 1. 自陣に十分近いか、敵陣側扇範囲内に捉えている味方がいないのいずれを満たす敵がいれば最も近い味方を割り振る。
683 パラメータはdPrioritizedAimLeaker,thetaPrioritizedAimLeakerの二つ
684 2. 距離の近い順に割り振る。
685 */
686 //まず、lostしてしまったものを削除
687 if(!target.is_none() && target.state==TrackInfo::UpdateState::LOST){
688 target=TrackInfo();
689 }
690 if(!launchedTarget.is_none() && launchedTarget.state==TrackInfo::UpdateState::LOST){
691 launchedTarget=TrackInfo();
692 }
693 std::vector<Eigen::Vector3d> fpos,fvel;
694 std::vector<bool> assigned;
695 assigned.push_back(false);
696 fpos.push_back(pos);
697 fvel.push_back(vel);
698 for(auto&& f:parent->observables.at("/shared/fighter"_json_pointer).items()){
699 if(f.value().at("isAlive").get<bool>()){
700 assigned.push_back(false);
701 MotionState mo(f.value().at("motion"));
702 fpos.push_back(mo.pos(getLocalCRS()));
703 fvel.push_back(mo.vel(getLocalCRS()));
704 }
705 }
706 std::size_t numF=fpos.size();
707 std::vector<TrackInfo*> candidates;
708 std::vector<Eigen::Vector3d> tpos;
709 for(auto& t:allEnemies){
710 if(t.state!=TrackInfo::UpdateState::LOST){
711 candidates.push_back(&t);
712 tpos.push_back(t.pos(getLocalCRS()));
713 }
714 }
715 isAssignedAsInterceptor=false;
716 if(candidates.size()==0){//何も見えていない場合
717 target=TrackInfo();
718 return;
719 }
720 //(1)突破阻止(複数該当した場合は、とりあえずラインに近い敵から順に、最も近い未割当の味方を割り当てる)
721 std::vector<std::pair<double,int>> leakers,nearestF;
722 Eigen::MatrixXd dist=Eigen::MatrixXd::Zero(fpos.size(),tpos.size());
723 int i=0;int j=0;
724 for(auto& tp:tpos){
725 double distFromLine=dLine-tp.dot(-forward);
726 bool isLeaker=distFromLine<dPrioritizedAimLeaker*dLine;
727 double angle=M_PI;
728 i=0;
729 for(auto& fp:fpos){
730 dist(i,j)=(fp-tp).norm();
731 angle=std::min(angle,
732 acos(std::max(-1.0,std::min(1.0,
733 (tp-fp).block<2,1>(0,0,2,1).normalized().dot(forward.block<2,1>(0,0,2,1))
734 )))
735 );
736 i++;
737 }
738 if(angle>thetaPrioritizedAimLeaker){
739 isLeaker=true;
740 }
741 if(isLeaker){
742 leakers.push_back(std::make_pair(distFromLine,j));
743 }
744 j++;
745 }
746 std::sort(leakers.begin(),leakers.end(),
747 [](const std::pair<double,int> &lhs,const std::pair<double,int> &rhs){
748 return lhs.first<rhs.first;
749 });
750 int numAssigned=0;
751 //typedef std::pair<double,int> viType;
752 for(auto& li:leakers){
753 nearestF.clear();
754 for(int i=0;i<fpos.size();++i){
755 nearestF.push_back(std::make_pair(dist(i,li.second),i));
756 }
757 std::sort(nearestF.begin(),nearestF.end(),
758 [](const std::pair<double,int> &lhs,const std::pair<double,int> &rhs){
759 return lhs.first<rhs.first;
760 });
761 for(auto& nf:nearestF){
762 if(!assigned[nf.second]){
763 numAssigned++;
764 if(nf.second==0){
765 target=*candidates[li.second];
766 isAssignedAsInterceptor=true;
767 }
768 break;
769 }
770 }
771 if(numAssigned==fpos.size()){
772 break;
773 }
774 }
775 //(2)最も近い敵(重複可)
776 std::vector<TrackInfo*> nearestE;
777 for(int i=0;i<fpos.size();++i){
778 int idx;
779 Eigen::VectorXd tmp=dist.block(i,0,1,tpos.size()).transpose();
780 tmp.minCoeff(&idx);
781 nearestE.push_back(candidates[idx]);
782 }
783 target=*nearestE[0];
784 return;

790bool BasicAACRuleBasedAgent01::chooseTransitionNormal(){
791 /*通常時の状態遷移(前進、接近、横行、後退)の選択を行う。状況に応じて確率で決定
792 1. 帰還できなくなりそうなとき、確定で後退
793 2. 見えている敵がいない場合、確定で前進
794 3. 敵陣に十分近いか、敵陣側扇範囲内に敵がいないのいずれかであればライン突破を優先し前進、
795 　主目標が突破阻止を目的として割り当てられているとき、突破阻止を優先し接近を選択する。
796 　両方満たす場合、自身が先に突破しそうであれば突破を優先し前進を選択、相手が先に突破しそうであれば突破阻止を優先し接近を選択
797 4. 主目標を味方が余裕をもって捉えていれば前進、接近、横行から確率で選択
798 5. 主目標を味方が余裕をもって捉えておらず、自身で余裕をもって捉えていなければ確定で接近
799 6. 主目標を味方が余裕をもって捉えておらず、自身で余裕をもって捉えていれば横行または接近から確率で選択
800 Returns:
801 bool: Trueのとき、もう一度回す。
802 */
803 const nl::json& fObs=parent->observables;
804 bool reconsider=manager->getTime()-lastNormalTransitionTime>=reconsiderationCycle;
805 //1. 帰還できなくなりそう
806 if(!chkFuelSufficiency()){
807 if(transitionCause!=TransitionCause::FUEL_SHORTAGE || reconsider){
808 transitionCause=TransitionCause::FUEL_SHORTAGE;
809 immidiateTransition(State::RTB);
810 }
811 return false;
812 }
813 //2. 見えている相手がいない
814 if(allEnemies.size()==0){
815 if(transitionCause!=TransitionCause::NO_ENEMY || reconsider){
816 transitionCause=TransitionCause::NO_ENEMY;
817 immidiateTransition(State::ADVANCE);
818 }
819 return false;
820 }
821 //3. ライン突破を優先 or 突破阻止を優先
822 double dist=dLine-pos.dot(forward);
823 double angle=M_PI;
824 bool tryBreak=false;
825 for(auto& t:allEnemies){
826 Eigen::Vector3d tpos=t.pos(getLocalCRS());
827 angle=std::min(angle,acos(std::max(-1.0,std::min(1.0,(tpos-pos).block<2,1>(0,0,2,1).normalized().dot(forward.block<2,1>(0,0,2,1))))));
828 }
829 tryBreak=dist<dPrioritizedAdvance*dLine || angle>thetaPrioritizedAdvance;
830 if(isAssignedAsInterceptor){
831 if(tryBreak){
832 double eDist=dLine-target.pos(getLocalCRS()).dot(-forward);
833 if(eDist<dist){
834 //相手の方が先に突破する可能性が高い場合、突破阻止を優先
835 if(transitionCause!=TransitionCause::INTERCEPT_LEAKER || reconsider){
836 transitionCause=TransitionCause::INTERCEPT_LEAKER;
837 immidiateTransition(State::APPROACH_TARGET);
838 }
839 return false;
840 }
841 }
842 }
843 if(tryBreak){
844 //自分の方が先に突破する可能性が高い場合、突破を優先
845 if(transitionCause!=TransitionCause::TRY_BREAK || reconsider){
846 transitionCause=TransitionCause::TRY_BREAK;
847 immidiateTransition(State::ADVANCE);
848 }
849 return false;
850 }
851 //4〜6
852 if(target.nonLimitTrackers.size()>=2){
853 //4. 味方が余裕をもって捉えている
854 if(transitionCause!=TransitionCause::TRACKING_BY_ALLY || reconsider){
855 std::vector<nl::json> candidates={State::ADVANCE,State::APPROACH_TARGET,State::KEEP_SENSING};
856 Eigen::VectorXd probs(3);
857 probs<<pAdvanceAlly,pApproachAlly,pKeepSensingAlly;
858 probs/=probs.sum();
859 nl::json config={
860 {"type","choice"},
861 {"weights",probs},
862 {"candidates",candidates}
863 };
864 auto nextState=getValueFromJsonR(config,randomGen).get<State>();
865 transitionCause=TransitionCause::TRACKING_BY_ALLY;
866 immidiateTransition(nextState);
867 }
868 return false;
869 }else if(target.nonLimitTrackers.size()==1){
870 if(target.nonLimitTrackers[0]!=parent->getFullName()){
871 //4. 味方が余裕をもって捉えている
872 if(transitionCause!=TransitionCause::TRACKING_BY_ALLY || reconsider){
873 std::vector<nl::json> candidates={State::ADVANCE,State::APPROACH_TARGET,State::KEEP_SENSING};
874 Eigen::VectorXd probs(3);
875 probs<<pAdvanceAlly,pApproachAlly,pKeepSensingAlly;
876 probs/=probs.sum();
877 nl::json config={
878 {"type","choice"},
879 {"weights",probs},
880 {"candidates",candidates}
881 };
882 auto nextState=getValueFromJsonR(config,randomGen).get<State>();
883 transitionCause=TransitionCause::TRACKING_BY_ALLY;
884 immidiateTransition(nextState);
885 }
886 return false;
887 }else{
888 //5. 自分だけが余裕をもって捉えている
889 if(transitionCause!=TransitionCause::TRACKING_BY_MYSELF || reconsider){
890 std::vector<nl::json> candidates={State::APPROACH_TARGET,State::KEEP_SENSING};
891 Eigen::VectorXd probs(2);
892 probs<<pApproachMyself,pKeepSensingMyself;
893 probs/=probs.sum();
894 nl::json config={
895 {"type","choice"},
896 {"weights",probs},
897 {"candidates",candidates}
898 };
899 auto nextState=getValueFromJsonR(config,randomGen).get<State>();
900 transitionCause=TransitionCause::TRACKING_BY_MYSELF;
901 immidiateTransition(nextState);
902 }
903 return false;
904 }
905 }else{
906 //6. 誰も余裕をもって捉えていない
907 if(transitionCause!=TransitionCause::TRACKING_BY_NOBODY || reconsider){
908 transitionCause=TransitionCause::TRACKING_BY_NOBODY;
909 immidiateTransition(State::APPROACH_TARGET);
910 }
911 return false;
912 }

1013void BasicAACRuleBasedAgent01::control(){
1014 /*stateに応じた制御出力を決定する。
1015 * stateに応じて進みたい方向dstDirと進みたい速さdstVを計算する。
1016 * dstDirに対しては、現時点の進行方向から導出される回転方向に回転するような角速度を発生させる。
1017 * dstVに対しては、現時点の速度から導出される加速度を発生させる。
1018 * 高度維持、マルチセンシングもこの関数内でdstDirに補正をかけることで行う。
1019 * 目標への接近はその瞬間に目標がいる方向へ最大速度とする。
1020 */
1021 if(!isAfterDeploy){
1022 updateMyInfo();//自機情報の更新
1023 updateMWS();//MWS情報の更新
1024 updateTracks(false);//航跡情報の更新
1025 updateTargetStatus(false);//目標情報の更新
1026 }
1027 isAfterDeploy=false;
1028 Eigen::Vector3d ex=myMotion.dirBtoP(Eigen::Vector3d(1,0,0));
1029 Eigen::Vector3d dstDirH;
1030 //水平方向の決定
1031 if(state==State::ADVANCE){
1032 //前進
1033 dstDirH=forward;
1034 }else if(state==State::APPROACH_TARGET){
1035 //その瞬間に目標がいる方向
1036 Eigen::Vector3d tpos=target.pos(getLocalCRS());
1037 dstDirH=tpos-pos;
1038 dstDirH(2)=0;
1039 dstDirH.normalize();
1040 }else if(state==State::KEEP_SENSING){
1041 //横行ははじめから複数捉えることを考慮する
1042 //主目標方向を正面として、左右どちらに進むかを決める
1043 //(1)場外に出そうならば内側向き(dOutLimitKeepSensing)
1044 //(2)場外に出ないならば副目標を多数捉えられる向きを優先
1045 //(3)副目標が同数ならば前進優先
1046 Eigen::Vector3d tDir=target.pos(getLocalCRS())-pos;
1047 double az=atan2(tDir(1),tDir(0));
1048 Eigen::Vector3d right=Eigen::Vector3d(cos(az+M_PI/2.-thetaKeepSensing),sin(az+M_PI/2.-thetaKeepSensing),0.);
1049 Eigen::Vector3d left=Eigen::Vector3d(cos(az-M_PI/2.+thetaKeepSensing),sin(az-M_PI/2.+thetaKeepSensing),0.);
1050 if(abs(pos(0))>=dOut-dOutLimitKeepSensing){
1051 //(1)
1052 Eigen::Vector3d inner;
1053 if(pos(0)>0){
1054 inner<<-1.,0.,0.;
1055 }else{
1056 inner<<1.,0.,0.;
1057 }
1058 double r=right.dot(inner);
1059 double l=left.dot(inner);
1060 if(r>=l){
1061 dstDirH=right;
1062 }else{
1063 dstDirH=left;
1064 }
1065 }else{
1066 //(2)
1067 int leftCount=0;
1068 int rightCount=0;
1069 if(parent->observables.contains("/spec/sensor/radar"_json_pointer)){
1070 for(auto& t:additionalTargets){
1071 Eigen::Vector3d tpos=t.pos(getLocalCRS());
1072 tDir=(tpos-pos).array()*Eigen::Array3d(1,1,0);
1073 double sAngle=parent->observables.at("/spec/sensor/radar/thetaFOR"_json_pointer);
1074 if(tDir.dot(right)>=cos(sAngle)){
1075 rightCount++;
1076 }
1077 if(tDir.dot(left)>=cos(sAngle)){
1078 leftCount++;
1079 }
1080 }
1081 }
1082 if(rightCount>leftCount){
1083 dstDirH=right;
1084 }else if(leftCount>rightCount){
1085 dstDirH=left;
1086 }else{
1087 //(3)
1088 double r=right.dot(forward);
1089 double l=left.dot(forward);
1090 if(r>=l){
1091 dstDirH=right;
1092 }else{
1093 dstDirH=left;
1094 }
1095 }
1096 }
1097 }else if(state==State::RTB){
1098 //後退
1099 dstDirH=-forward;
1100 }else if(state==State::WITHDRAW){
1101 //基本的には離脱のきっかけとなった敵から最も離れる方位に向かう
1102 std::vector<Eigen::Vector3d> tpos;
1103 for(auto& t:allEnemies){
1104 for(auto& trig:withdrawnTrigger){
