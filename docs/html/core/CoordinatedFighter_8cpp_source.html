153double SimpleFighterJetEngine::calcThrust(double power_,double alt,double rmach) const{
154 Eigen::VectorXd x{{rmach,m2ft(alt)}};
155 double tmin=tminTable(x);
156 double tmil=tmilTable(x);
157 double tmax=tmaxTable(x);
158 if(power_<0.5){
159 return lbf2N(tmin+(tmil-tmin)*(power_/0.5));
160 }else{
161 return lbf2N(tmil+(tmax-tmil)*(power_/0.5-1.0));
162 }

164double SimpleFighterJetEngine::trimPower(double tgtThrust,double alt,double Mach) const{
165 double thrustMin=calcThrust(0.0,alt,Mach);
166 double thrustMax=calcThrust(1.0,alt,Mach);
167 auto sub=[&](double pCmd){
168 return calcThrust(pCmd,alt,Mach)-tgtThrust;
169 };
170 double trimmed;
171 if(thrustMin>=tgtThrust){
172 trimmed=0.0;
173 }else if(thrustMax<=tgtThrust){
174 trimmed=1.0;
175 }else{
176 boost::math::tools::eps_tolerance<double> tol(16);
177 boost::uintmax_t max_iter=10;
178 try{
179 auto result=boost::math::tools::toms748_solve(sub,0.0,1.0,tol,max_iter);
180 trimmed=(result.first+result.second)/2;
181 }catch(std::exception& e){
182 std::cout<<"at trimPower"<<std::endl;
183 DEBUG_PRINT_EXPRESSION(sub(0.0))
184 DEBUG_PRINT_EXPRESSION(sub(1.0))
185 DEBUG_PRINT_EXPRESSION(tgtThrust)
186 DEBUG_PRINT_EXPRESSION(alt)
187 DEBUG_PRINT_EXPRESSION(Mach)
188 throw e;
189 }
190 }
191 return trimmed;

211double SimpleFighterJetEngine::calcdTdH(double power_,double alt,double rmach) const{
212 Eigen::VectorXd x{{rmach,m2ft(alt)}};
213 double tmin_grad=tminTable.grad(x)(1);
214 double tmil_grad=tmilTable.grad(x)(1);
215 double tmax_grad=tmaxTable.grad(x)(1);
216 if(power_<0.5){
217 return lbf2N(2*power_*m2ft(tmil_grad-tmin_grad));
218 }else{
219 return lbf2N((2*power_-1)*m2ft(tmax_grad-tmil_grad));
220 }
221}
222double SimpleFighterJetEngine::calcdTdM(double power_,double alt,double rmach) const{
223 Eigen::VectorXd x{{rmach,m2ft(alt)}};
224 double tmin_grad=tminTable.grad(x)(0);
225 double tmil_grad=tmilTable.grad(x)(0);
226 double tmax_grad=tmaxTable.grad(x)(0);
227 if(power_<0.5){
228 return lbf2N(2*power_*(tmil_grad-tmin_grad));
229 }else{
230 return lbf2N((2*power_-1)*(tmax_grad-tmil_grad));

233double SimpleFighterJetEngine::rtau(double dp) const{
234 if(dp<=0.25){
235 return 1.0;
236 }else if(dp>=0.50){
237 return 0.1;
238 }else{
239 return 1.9-3.6*dp;
240 }

242double SimpleFighterJetEngine::pdot(double pCurr_,double pCmd_) const{
243 double p2,T;
244 if(pCmd_>=0.50){
245 if(pCurr_>=0.50){
246 T=5.0;
247 p2=pCmd_;
248 }else{
249 p2=0.60;
250 T=rtau(p2-pCurr_);
251 }
252 }else{
253 if(pCurr_>=0.50){
254 T=5.0;
255 p2=0.40;
256 }else{
257 p2=pCmd_;
258 T=rtau(p2-pCurr_);
259 }
260 }
261 return T*(p2-pCurr_);

263void SimpleFighterJetEngine::behave(){
264 BaseType::behave();
265 if(!enableInstantResponse){
266 double alt=motion.getHeight();
267 std::map<std::string,double> atm=atmosphere(alt);
268 double V=motion.vel(manager->getRootCRS()).norm();
269 double Mach=V/atm["a"];
270 double dt=interval[SimPhase::BEHAVE]*manager->getBaseTimeStep();
271
272 std::function<Eigen::VectorXd (const Eigen::VectorXd& x,const Time& t)> deriv=[&](const Eigen::VectorXd& x,const Time& t){
273 Eigen::VectorXd f(1);
274 f(0)=pdot(std::clamp<double>(x(0),0,1),pCmd);
275 return f;
276 };
277 Eigen::VectorXd x0(1);
278 x0(0)=power;
279 Eigen::VectorXd nextX=RK4(x0,manager->getTime(),dt,deriv);
280 power=std::clamp<double>(nextX(0),0,1);
281 thrust=calcThrust(power,alt,Mach);
282 }

389double CoordinatedFighter::calcOptimumCruiseFuelFlowRatePerDistance(){
390 //high computational cost, so once you obtained the value, should write it in json modelConfig.
391 auto eng=getShared<SimpleFighterJetEngine>(engine);
392 double ffr=eng->getFuelFlowRate();
393 if(ffr<=0.0){
394 //no fuel consumption model
395 return 0.0;
396 }else{
397 Eigen::Vector3d gravityVector = motion.dirHtoP(Eigen::Vector3d(0,0,1),Eigen::Vector3d(0,0,0),"NED",false)*m*gravity; //TODO 慣性力モデルに置換。
398 std::function<double(unsigned int,const double*,double*,void*)> obj=[&](unsigned int n,const double* x,double* grad,void* data){
399 //x=[alt,V] in [m,m/s]
400 double alt=x[0];
401 double V=x[1];
402 std::map<std::string,double> atm=atmosphere(alt);
403 double Mach=V/atm["a"];
404 //trim α for level flight
405 double dp=atm["rho"]*V*V/2;
406 double tgtCz=-relPtoB(gravityVector)(2)/(dp*S);
407 double aoa_=trimAoa(tgtCz);
408 Eigen::Vector3d CFb=calcAero(aoa_,Mach);
409 double tgtThrust=-CFb(0)*dp*S;
410 double pCmd_=eng->trimPower(tgtThrust,alt,Mach);
411 double fuelFlowRate=eng->calcFuelFlowRate({{"power",pCmd_}});
412 if(!(
413 abs(tgtThrust-eng->calcThrust(pCmd_,alt,Mach))<1.0 &&
414 minCz<=tgtCz &&
415 maxCz>=tgtCz
416 )){//infeasible to keep steady level flight
417 return eng->calcFuelFlowRate({{"power",1.0}});
418 }
419 return fuelFlowRate/V;//fuel consumption per distance in [kg/m]
420 };
421 nlopt_delegator obj_d(&obj,nullptr);
422 std::vector<double> x={10000.0,300.0};
423 std::vector<double> lb={0.0,1.0};
424 std::vector<double> ub={20000.0,600.0};
425 nlopt::opt opt=nlopt::opt(nlopt::algorithm::LN_NELDERMEAD,2);
426 opt.set_min_objective(nlopt_delegatee,&obj_d);
427 opt.set_maxeval(1e5);
428 opt.set_lower_bounds(lb);
429 opt.set_upper_bounds(ub);
430 auto result=opt.optimize(x,ffr);
431 std::map<std::string,double> atm=atmosphere(x[0]);
432 double Mach=x[1]/atm["a"];
433 std::cout<<"optimize cruise: alt="<<x[0]<<"[m], V="<<x[1]<<"[m/s] (M="<<Mach<<")"<<", ffr="<<ffr<<"[kg/m] ("<<ffr*x[1]<<"[kg/s])"<<std::endl;
434 return ffr;
435 }

437double CoordinatedFighter::trimAoa(double tgtCz){
438 double trimmed;
439 if(minCz>=tgtCz){
440 trimmed=maxAoa;
441 }else if(maxCz<=tgtCz){
442 trimmed=minAoa;
443 }else{
444 auto sub=[&](double aoa_){
445 return calcAero(aoa_,0.0)(2)-tgtCz;
446 };
447 boost::math::tools::eps_tolerance<double> tol(16);
448 boost::uintmax_t max_iter=10;
449 try{
450 auto result=boost::math::tools::toms748_solve(sub,minAoa,maxAoa,tol,max_iter);
451 trimmed=(result.first+result.second)/2;
452 }catch(std::exception& e){
453 std::cout<<"at trimAoa"<<std::endl;
454 DEBUG_PRINT_EXPRESSION(sub(minAoa))
455 DEBUG_PRINT_EXPRESSION(sub(maxAoa))
456 DEBUG_PRINT_EXPRESSION(tgtCz)
457 throw e;
458 }
459 }
460 return trimmed;

500double CoordinatedFighter::cxa(double aoa_){
501 double x=rad2deg(aoa_);
502 double de_=trimmedDe(x);
503 double ddeda_=trimmedDe.grad(x);
504 Eigen::VectorXd grad=cxTable.grad(Eigen::VectorXd{{rad2deg(de_),x}});
505 double ret=(grad(1)+grad(0)*ddeda_)*(180/M_PI);
506 return ret;

513 return ret;
514}
515double CoordinatedFighter::cl(double aoa_){
516 return cx(aoa_)*sin(aoa_)-cz(aoa_)*cos(aoa_);
517}
518double CoordinatedFighter::cd(double aoa_,double mach_){
519 return -cx(aoa_)*cos(aoa_)-cz(aoa_)*sin(aoa_)+cdw(mach_);
520}
521double CoordinatedFighter::cdw(double mach_){
522 //wave drag is based on [3] and [4].
523 //coefficients are obtained by curve fitting with some points.
524 double cdw0=cdwTable(0);
525 double kdwm=cdwTable(1);
526 double kdw=cdwTable(2);
527 double dm=cdwTable(3);
528 double fm=1.0/(1.0+exp(-8*(mach_-(1.0-dm/2.0))/dm));
529 return fm*cdw0*kdw/pow(pow(pow(mach_-kdwm,2)-1.0,2)+pow(kdw,4),0.25);
530}
531Eigen::Vector3d CoordinatedFighter::calcAero(double aoa_,double mach_){
532 /*
533 "Coordinated" flight assumes de=da=dr=β=0.
534 Angle of attack is only considered for aerodynamic force and the aircraft attitude neglects it.
535 CX=-CD=Cx(α)*cos(α)+Cz(α)*sin(α)
536 CZ=-CL=Cz(α)*cos(α)-Cx(α)*sin(α)
537 Polynomial coefficients of Cx and Cz is based on [2] and the lift is maximized when α=0.66449rad
538 */
539 double Cx=cx(aoa_);
540 double Cz=cz(aoa_);
541 return Eigen::Vector3d(Cx*cos(aoa_)+Cz*sin(aoa_)-cdw(mach_),0.0,Cz*cos(aoa_)-Cx*sin(aoa_));
542}
543void CoordinatedFighter::calcMotion(double dt){
544 motion_prev=motion;
545 Quaternion qPrev=motion.q;
546 Eigen::Vector3d gravityVector = motion.dirHtoP(Eigen::Vector3d(0,0,1),Eigen::Vector3d(0,0,0),"NED",false)*m*gravity; //TODO 慣性力モデルに置換。
547 double alt=motion.getHeight();
548 double V=motion.vel.norm();
549 std::map<std::string,double> atm=atmosphere(alt);
550 double Mach=V/atm["a"];
551 double dp=atm["rho"]*V*V/2;
552 nl::json ctrl=controllers["FlightController"].lock()->commands["motion"];
553 double thrust=getThrust();
554 double aoa=ctrl.at("aoa");
555 double dRoll=ctrl.at("dRoll");
556 Eigen::Vector3d rollAx=motion.vel()/V;
557 //Roll rotation(Immidiate)
558 if(abs(dRoll)>1e-8){
559 Quaternion dq1=Quaternion::fromAngle(rollAx,dRoll);
560 motion.setQ(dq1*motion.q);
561 }else{
562 Eigen::VectorXd dq1=motion.q.dqdwi()*rollAx*dRoll;
563 motion.setQ((motion.q+Quaternion(dq1)).normalized());
564 }
565 //position and velocity update
566 Eigen::Vector3d CFb=calcAero(aoa,Mach);
567 Eigen::Vector3d F=relBtoP(Eigen::Vector3d(thrust*cos(aoa),0.,-thrust*sin(aoa))+CFb*(dp*S))+gravityVector;
568 Eigen::Vector3d accel=F/m;
569 double accelScalar=accel.dot(rollAx);
570 Eigen::Vector3d omegaV=rollAx.cross(accel/V);
571 double rotAngle=omegaV.norm()*dt;
572 if(rotAngle<1e-8){
573 motion.setPos(motion.pos()+motion.vel()*dt+accel*dt*dt/2.);
574 motion.setVel(motion.vel()+accel*dt);
575 Eigen::VectorXd dq2=motion.q.dqdwi()*omegaV*dt;
576 motion.setQ((motion.q+Quaternion(dq2)).normalized());
577 }else{
578 Eigen::Vector3d ex0=motion.vel()/V;
579 Eigen::Vector3d ey0=omegaV.cross(ex0).normalized();
580 double wt=rotAngle;
581 double acc=accelScalar;
582 Eigen::Vector3d vn=(ex0*cos(wt)+ey0*sin(wt)).normalized();
583 motion.setVel((V+acc*dt)*vn);
584 Eigen::Vector3d dr=V*dt*(ex0*sinc(wt)+ey0*oneMinusCos_x2(wt)*wt)+acc*dt*dt*(ex0*(sinc(wt)-oneMinusCos_x2(wt))+ey0*wt*(sincMinusOne_x2(wt)+oneMinusCos_x2(wt)));
585 motion.setPos(motion.pos()+dr);
586 accel=(motion.vel()-vel_prev())/dt;
587 Quaternion dq2=Quaternion::fromAngle(omegaV/(rotAngle/dt),rotAngle);
588 motion.setQ(dq2*motion.q);
589 }
590 Quaternion dq=motion.q*qPrev.conjugate();
591 Eigen::Vector3d ax=motion.q.vec();
592 double sn=ax.norm();
593 if(sn>0){ax/=sn;}
594 motion.setOmega(ax*atan2(sn,dq.w())+rollAx*dRoll/dt);
595}

654nl::json CoordinatedFlightController::calcDirect(const nl::json &cmd){
655 auto p=getShared<CoordinatedFighter>(parent);
656 auto roll=std::clamp(cmd.at("roll").get<double>(),-1.,1.);
657 auto pitch=std::clamp(cmd.at("pitch").get<double>(),-1.,1.);
658 double alt=p->motion.getHeight();
659 double V=p->motion.vel.norm();
660 std::map<std::string,double> atm=atmosphere(alt);
661 double Mach=V/atm["a"];
662 double dp=atm["rho"]*V*V/2;
663 Eigen::Vector3d down=p->motion.dirHtoP(Eigen::Vector3d(0,0,1),Eigen::Vector3d(0,0,0),"NED",false);
664 Eigen::Vector3d gravityVector = down*p->m*gravity; //TODO 慣性力モデルに置換。
665 double tgtCz=-p->relPtoB(gravityVector)(2)/(dp*p->S);
666 double aoa_=std::clamp(p->trimAoa(tgtCz)+(p->maxAoa-p->minAoa)*pitch/2,p->minAoa,p->maxAoa);
667 Eigen::Vector3d CFb=p->calcAero(aoa_,Mach);
668 double tgtThrust=-CFb(0)*dp*p->S;
669 auto eng=getShared<SimpleFighterJetEngine>(p->engine);
670 double pCmd_=eng->trimPower(tgtThrust,alt,Mach);
671 if(cmd.contains("throttle")){//0〜1
672 pCmd_=std::clamp(cmd.at("throttle").get<double>(),0.,1.);
673 }else if(cmd.contains("accel")){//-1〜+1
674 double clampedCmd=std::clamp(cmd.at("accel").get<double>(),-1.,1.);
675 if(clampedCmd>=0){
676 pCmd_=pCmd_+(1.0-pCmd_)*clampedCmd;
677 }else{
678 pCmd_=pCmd_*(1+clampedCmd);
679 }
680 }else{
681 throw std::runtime_error("Either 'throttle' or 'accel' must be in cmd keys.");
682 }
683 double dRoll_=p->rollMax*roll*interval[SimPhase::BEHAVE]*manager->getBaseTimeStep();
684 //side G Limitter
685 Eigen::Vector3d ey=p->relBtoP(Eigen::Vector3d(0,cos(dRoll_),sin(dRoll_)));
686 Eigen::Vector3d ez=p->relBtoP(Eigen::Vector3d(0,-sin(dRoll_),cos(dRoll_)));
687 double zz=p->calcAero(aoa_,Mach)(2)/(p->m*gravity)+ez.dot(down);
688 if(Eigen::Vector3d(0,ey.dot(down),zz).norm() >= p->sideGLimit){
689 if(zz>=0){
690 tgtCz=p->m*util::gravity*(sqrt(p->sideGLimit-ey.dot(down)*ey.dot(down))-ez.dot(down));
691 }else{
692 tgtCz=-p->m*util::gravity*(sqrt(p->sideGLimit-ey.dot(down)*ey.dot(down))-ez.dot(down));
693 }
694 aoa_=p->trimAoa(tgtCz);
695 }
696 return {{"pCmd",pCmd_},{"aoa",aoa_},{"dRoll",dRoll_}};
697}
698nl::json CoordinatedFlightController::calcFromDirAndVel(const nl::json &cmd){
699 auto p=getShared<CoordinatedFighter>(parent);
700 double eps=1e-4;
701 double alt=p->motion.getHeight();
702 double V=p->motion.vel.norm();
703 Eigen::Vector3d d=p->motion.vel()/V;
704 std::map<std::string,double> atm=atmosphere(alt);
705 double Mach=V/atm["a"];
706 double dp=atm["rho"]*V*V/2;
707 Eigen::Vector3d gravityVector = p->motion.dirHtoP(Eigen::Vector3d(0,0,1),Eigen::Vector3d(0,0,0),"NED",false)*p->m*gravity; //TODO 慣性力モデルに置換。
708 Eigen::Vector3d gravityVectorVert=gravityVector-(gravityVector.dot(d))*d;//==g.dot(n)+g.dot(b) for any φ
709 Eigen::Vector3d b,n;
710 auto eng=getShared<SimpleFighterJetEngine>(p->engine);
711 double Tmin=eng->calcThrust(0.0,alt,Mach);
712 double Tmax=eng->calcThrust(1.0,alt,Mach);
713 double dstVd,dstP;//velocity command
714 double dstTd,dstAoa;//turn command
715 int velCmdType,turnCmdType;
716 if(cmd.contains("dstV")){
717 velCmdType=0;
718 double dstV=cmd.at("dstV");
719 dstVd=-lambdaVel*(V-dstV);
720 }else if(cmd.contains("dstAccel")){
721 velCmdType=1;
722 double dstAccel=cmd.at("dstAccel");
723 dstVd=dstAccel;
724 }else if(cmd.contains("dstThrust")){
725 velCmdType=2;
726 double dstT=cmd.at("dstThrust");
727 dstP=std::clamp((dstT-Tmin)/(Tmax-Tmin),0.0,1.0);
728 }else if(cmd.contains("dstThrottle")){
729 velCmdType=2;
730 double dstT=eng->calcThrust(cmd.at("dstThrottle"),alt,Mach);
731 dstP=std::clamp((dstT-Tmin)/(Tmax-Tmin),0.0,1.0);
732 }else{
733 throw std::runtime_error("Only one of dstV, dstAccel, dstThrust or dstThrottle is acceptable.");
734 }
735 if(cmd.contains("dstDir")){
736 turnCmdType=0;
737 Eigen::Vector3d dstDir=cmd.at("dstDir");
738 if(cmd.contains("dstAlt")){
739 dstDir=altitudeKeeper(p->motion,dstDir,cmd.at("dstAlt").get<double>());
740 }
741 dstDir.normalize();
742 b=d.cross(dstDir);
743 double snTheta=std::min(1.0,b.norm());
744 double csTheta=std::clamp(d.dot(dstDir),-1.0,1.0);
745 double eTheta=atan2(snTheta,csTheta);
746 if(snTheta<eps){
747 //Almost no rotation or 180deg rotation
748 b=gravityVectorVert.normalized();
749 dstTd=lambdaTheta*std::min(eTheta,clampTheta);
750 }else{
751 b/=snTheta;
752 //ピッチ制限の判定
753 if(eTheta>=pitchLimitThreshold){//一定以上の回転指示
754 b=pitchLimitter(p->motion,dstDir,pitchLimit);
755 }
756 dstTd=lambdaTheta*std::min(eTheta,clampTheta);
757 }
758 }else if(cmd.contains("dstTurnRate")){
759 turnCmdType=1;
760 Eigen::Vector3d dstTurnRate=cmd.at("dstTurnRate");
761 if(dstTurnRate.norm()<eps){
762 //Almost no rotation
763 b=gravityVectorVert.normalized();
764 dstTd=0;
765 }else{
766 b=d.cross(dstTurnRate.cross(d)).normalized();
767 dstTd=b.dot(dstTurnRate);
768 }
769 }else if(cmd.contains("dstAlpha") && cmd.contains("ey")){
770 turnCmdType=2;
771 dstAoa=std::clamp<double>(cmd.at("dstAlpha"),p->minAoa,p->maxAoa);
772 b=d.cross((cmd.at("ey").get<Eigen::Vector3d>()).cross(d)).normalized();
773 }else{
774 throw std::runtime_error("Only one of dstDir, dstTurnRate, or (dstAlpha & ey) is acceptable.");
775 }
776 n=b.cross(d);
777 double gt=gravityVector.dot(d)/p->m;
778 double gb_v=gravityVector.dot(b)/(p->m*V);
779 double gn_v=gravityVector.dot(n)/(p->m*V);
780 std::function<double(unsigned int,const double*,double*,void*)> obj=[&](unsigned int n,const double* x,double* grad,void* data){//x=[T,alpha]
781 //x[0]:power [0,1)
782 //x[1]:aoa [minAoa,maxAoa]
783 //x[2]:phi [-pi,pi)
784 double ct=(Tmin+x[0]*(Tmax-Tmin));
785 double aoa=x[1];
786 double phi=x[2];
787 double dps=dp*p->S;
788 double cx_=p->cx(aoa)*dps;
789 double cz_=p->cz(aoa)*dps;
790 double ctp=Tmax-Tmin;
791 double cxa_=p->cxa(aoa)*dps;
792 double cza_=p->cza(aoa)*dps;
793 double ft=((ct+cx_)*cos(aoa)+cz_*sin(aoa))/p->m;
794 double ft_a=(-cx_+cza_)*sin(aoa)+(cz_+cxa_)*cos(aoa)/p->m;
795 double fnb=((ct+cx_)*sin(aoa)-cz_*cos(aoa))/(p->m*V);
796 double fnb_a=((cz_+cxa_)*sin(aoa)+(cx_-cza_)*cos(aoa))/(p->m*V);
797 double vd=ft-(p->cdw(Mach)*dps/p->m)+gt;
798 double td=fnb*cos(phi)+gn_v;
799 double ed=fnb*sin(phi)+gb_v;
800 double f=0;
801 Eigen::Vector3d g=Eigen::Vector3d::Zero();
802 if(velCmdType==0){//dstV
803 double fv1=vd-dstVd;
804 f+=kVel*fv1*fv1;
805 if(grad){
806 g+=2*kVel*fv1*Eigen::Vector3d(ctp*cos(aoa)/p->m,ft_a,0);
807 }
808 }else if(velCmdType==1){//dstAccel
809 double fv1=vd-dstVd;
810 f+=kAccel*fv1*fv1;
811 if(grad){
812 g+=2*kAccel*fv1*Eigen::Vector3d(ctp*cos(aoa)/p->m,ft_a,0);
813 }
814 }else{//dstThrust or dstThrottle
815 double fv1=x[0]-dstP;
816 f+=kPower*fv1*fv1;
817 if(grad){
818 g+=Eigen::Vector3d(2*kPower*fv1,0,0);
819 }
820 }
821 if(turnCmdType==0){//dstDir
822 double ft1=td-dstTd;
823 f+=kTheta*ft1*ft1;
824 if(grad){
825 g+=2*kTheta*ft1*Eigen::Vector3d(ctp*sin(aoa)*cos(phi)/(p->m*V),fnb_a*cos(phi),-fnb*sin(phi));
826 }
827 }else if(turnCmdType==1){//dstOmega
828 double ft1=td-dstTd;
829 f+=kOmega*ft1*ft1;
830 if(grad){
831 g+=2*kOmega*ft1*Eigen::Vector3d(ctp*sin(aoa)*cos(phi)/(p->m*V),fnb_a*cos(phi),-fnb*sin(phi));
832 }
833 }else{//dstAlpha
834 double ft1=x[1]-dstAoa;
835 f+=kAoa*ft1*ft1;
836 if(grad){
837 g+=Eigen::Vector3d(0,2*kAoa*ft1,0);
838 }
839 }
840 f+=kEps*ed*ed;
841 if(grad){
842 g+=kEps*Eigen::Vector3d(ctp*sin(aoa)*sin(phi)/(p->m*V),fnb_a*sin(phi),fnb*cos(phi));
843 grad[0]=g(0);
844 grad[1]=g(1);
845 grad[2]=g(2);
846 }
847 return f;
848 };
849 std::function<double(unsigned int,const double*,double*,void*)> cnstr=[&](unsigned int n,const double* x,double* grad,void* data){//x=[T,alpha]
850 //x[0]:power [0,1)
851 //x[1]:aoa [minAoa,maxAoa]
852 //x[2]:phi [-pi,pi)
853 double ct=(Tmin+x[0]*(Tmax-Tmin));
854 double aoa=x[1];
855 double phi=x[2];
856 double dps=dp*p->S;
857 double cx_=p->cx(aoa)*dps;
858 double cz_=p->cz(aoa)*dps;
859 double ctp=Tmax-Tmin;
860 double cxa_=p->cxa(aoa)*dps;
861 double cza_=p->cza(aoa)*dps;
862 double fnb=((ct+cx_)*sin(aoa)-cz_*cos(aoa))/(p->m*V);
863 double fnb_a=((cz_+cxa_)*sin(aoa)+(cx_-cza_)*cos(aoa))/(p->m*V);
864 double td=fnb*cos(phi)+gn_v;
865 double ed=fnb*sin(phi)+gb_v;
866 double f=(td*td+ed*ed)*V*V-pow(util::gravity*p->sideGLimit,2);
867 if(grad){
868 Eigen::Vector3d g=Eigen::Vector3d(
869 ctp*sin(aoa)/(p->m*V)*(td*cos(phi)+ed*sin(phi)),
870 fnb_a*(td*cos(phi)+ed*sin(phi)),
871 fnb*(-td*sin(phi)+ed*cos(phi))
872 )*(2*V*V);
873 grad[0]=g(0);
874 grad[1]=g(1);
875 grad[2]=g(2);
876 }
877 return f;
878 };
879 nlopt_delegator obj_d(&obj,nullptr);
880 nlopt_delegator cnstr_d(&cnstr,nullptr);
881 std::vector<double> x_p={0.5,(p->minAoa+p->maxAoa)/2,0.0};
882 std::vector<double> x_n={0.5,(p->minAoa+p->maxAoa)/2,M_PI};
883 std::vector<double> lb_p={0.0,p->minAoa,-M_PI_2};
884 std::vector<double> ub_p={1.0,p->maxAoa,M_PI_2};
885 std::vector<double> lb_n={0.0,p->minAoa,M_PI_2};
886 std::vector<double> ub_n={1.0,p->maxAoa,3*M_PI_2};
887 nlopt::opt opt=nlopt::opt(nlopt::algorithm::LD_MMA,3);
888 opt.set_min_objective(nlopt_delegatee,&obj_d);
889 opt.add_inequality_constraint(nlopt_delegatee,&cnstr_d);
890 opt.set_xtol_rel(1e-4);
891 opt.set_ftol_rel(1e-4);
892 opt.set_xtol_abs(1e-4);
893 opt.set_ftol_abs(1e-4);
894 opt.set_maxeval(200);
895 double minf_p,minf_n;
896 try{
897 opt.set_lower_bounds(lb_p);
898 opt.set_upper_bounds(ub_p);
899 auto result=opt.optimize(x_p,minf_p);
900 }catch(...){
901 if(std::isnan(x_p[0])){x_p[0]=1.0;}
902 if(std::isnan(x_p[1])){x_p[1]=p->trimAoa(-p->m*gravity);}
903 if(std::isnan(x_p[2])){x_p[2]=atan2(-gn_v,-gb_v);}
904 minf_p=std::numeric_limits<double>::infinity();
905 }
906 try{
907 opt.set_lower_bounds(lb_n);
908 opt.set_upper_bounds(ub_n);
909 auto result=opt.optimize(x_n,minf_n);
910 }catch(...){
911 if(std::isnan(x_n[0])){x_n[0]=1.0;}
912 if(std::isnan(x_n[1])){x_n[1]=p->trimAoa(-p->m*gravity);}
913 if(std::isnan(x_n[2])){x_n[2]=atan2(-gn_v,-gb_v);}
914 minf_n=std::numeric_limits<double>::infinity();
915 }
916 double thrust,aoa,phi;
917 Eigen::Vector3d eyPrev=p->relBtoP(Eigen::Vector3d(0.,1.,0.));
918 if(abs(x_n[2]-x_p[2]-M_PI)<0.05 && abs(x_n[1]-x_p[1])<0.05){
919 Eigen::Vector3d ey_p=(-n*sin(x_p[2])+b*cos(x_p[2])).normalized();
920 Eigen::Vector3d ey_n=(-n*sin(x_n[2])+b*cos(x_n[2])).normalized();
921 if(eyPrev.dot(ey_p)>=eyPrev.dot(ey_n)){
922 thrust=Tmin+x_p[0]*(Tmax-Tmin);
923 aoa=x_p[1];
924 phi=x_p[2];
925 }else{
926 thrust=Tmin+x_n[0]*(Tmax-Tmin);
927 aoa=x_n[1];
928 phi=x_n[2];
929 }
930 }else{
931 if(minf_p<minf_n){
932 thrust=Tmin+x_p[0]*(Tmax-Tmin);
933 aoa=x_p[1];
934 phi=x_p[2];
935 }else{
936 thrust=Tmin+x_n[0]*(Tmax-Tmin);
937 aoa=x_n[1];
938 phi=x_n[2];
939 }
940 }
941 Eigen::Vector3d ey=(-n*sin(phi)+b*cos(phi)).normalized();
942 double pCmd=eng->trimPower(thrust,alt,Mach);
943 double csDRoll=eyPrev.dot(ey);
944 double snDRoll=d.dot(eyPrev.cross(ey));
945 double dRoll=atan2(snDRoll,csDRoll);
946 return {{"pCmd",pCmd},{"aoa",aoa},{"dRoll",dRoll}};
947}
