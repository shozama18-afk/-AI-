227nl::json StructuredEntity::getNewComponentConfig(bool isManaged_, const std::string& newComponentName, const nl::json& newComponentConfig){
228 std::string baseName=newComponentConfig.at("baseName");
229
230 nl::json ic = newComponentConfig.contains("instanceConfig") ? newComponentConfig.at("instanceConfig") : nl::json::object();
231 ic["entityFullName"]=getFullName()+"/"+newComponentName;
232 ic["seed"]=randomGen();
233 ic["root"]=root(isManaged_);
234 ic["holder"]=non_const_this();
235
236 if(baseName=="PlaceHolder"){
237 return {
238 {"isEpisodic",isEpisodic()},
239 {"baseName",baseName},
240 {"className","PlaceHolderEntity"},
241 {"modelName",""},
242 {"modelConfig",nl::json::object()},
243 {"instanceConfig",std::move(ic)}
244 };
245 }
246
247 auto [cn,mn,mc,chain] = resolveModelConfig(newComponentConfig);
248
249 if(isManaged_){
250 // managedの場合、firstTick,intervalが未設定なら生成者(holder)と同じとする。
251 if(!mc.contains("firstTick") && !ic.contains("firstTick")){
252 if(modelConfig.contains("firstTick")){
253 mc["firstTick"]=modelConfig.at("firstTick");
254 }
255 }
256 if(!mc.contains("interval") && !ic.contains("interval")){
257 if(modelConfig.contains("interval")){
258 mc["interval"]=modelConfig.at("interval");
259 }
260 }
261 }else{
262 // unmanagedならつねに生成者(holder)と同じfirstTick,intervalを設定する。
263 if(modelConfig.contains("firstTick")){
264 mc["firstTick"]=modelConfig.at("firstTick");
265 }else{
266 if(mc.contains("firstTick")){
267 mc.erase("firstTick");
268 }
269 }
270 if(modelConfig.contains("interval")){
271 mc["interval"]=modelConfig.at("interval");
272 }else{
273 if(mc.contains("interval")){
274 mc.erase("interval");
275 }
276 }
277 }
278
279 return {
280 {"isEpisodic",isEpisodic()},
281 {"baseName",baseName},
282 {"className",cn},
283 {"modelName",mn},
284 {"modelConfig",std::move(mc)},
285 {"instanceConfig",std::move(ic)}
286 };
287}

309void StructuredEntity::registerComponent(bool inManagedTree,const std::shared_ptr<StructuredEntity>& root, const std::shared_ptr<StructuredEntity>& component){
310 std::string path=std::filesystem::path(component->getFullName()).lexically_relative(root->getFullName()).generic_string();
311 if(inManagedTree){
312 assert(component->isManaged());
313 root->_managedComponents.put(path,component);
314 }else{
315 assert(root==component || !component->isManaged());
316 root->_unmanagedComponents.put(path,component);
317 for(auto&& phase : root->getUnmanagedComponentPhases()){
318 root->_unmanagedComponentDependencyGraph[phase].add_instance(component);
319 }
320 }
