22bool checkSphericalAxisOrder(const std::string& order){
23 // Azimuth, Elevation, Distance(Range)の3種類の軸を含んでいるかどうかを判定する。
24 if(order.size()!=3){
25 return false;
26 }
27 bool found[3] = {false,false,false}; // Azimuth, Elevation, Distance(Range)
28 for(int i=0; i<3; ++i){
29 if(order.at(i)=='A'){ // Azimuth
30 found[0]=true;
31 }else if(order.at(i)=='E'){ // Elevation
32 found[1]=true;
33 }else if(order.at(i)=='D'){ // Distance
34 found[2]=true;
35 }else if(order.at(i)=='R'){ // Range
36 found[2]=true;
37 }
38 }
39 return found[0] && found[1] && found[2];
40}
41bool checkBodyCartesianAxisOrder(const std::string& order){
42 // FWD/AFT, STBD/PORT, UP/DOWNの3種類の軸を含んでいるかどうかを判定する。
43 // 左右を表すSTBD/PORTはRIGHT(R)/LEFT(L)も許容する。
44 if(order.size()!=3){
45 return false;
46 }
47 bool found[3] = {false,false,false}; // FWD/AFT, STBD/PORT, UP/DOWN
48 for(int i=0; i<3; ++i){
49 if(order.at(i)=='F'){ // FWD
50 found[0]=true;
51 }else if(order.at(i)=='A'){ // AFT
52 found[0]=true;
53 }else if(order.at(i)=='S'){ // STBD(RIGHT)
54 found[1]=true;
55 }else if(order.at(i)=='P'){ // PORT(LEFT)
56 found[1]=true;
57 }else if(order.at(i)=='R'){ // STBD(RIGHT)
58 found[1]=true;
59 }else if(order.at(i)=='L'){ // PORT(LEFT)
60 found[1]=true;
61 }else if(order.at(i)=='U'){ // UP
62 found[2]=true;
63 }else if(order.at(i)=='D'){ // DOWN
64 found[2]=true;
65 }
66 }
67 return found[0] && found[1] && found[2];
68}
69bool checkTopocentricCartesianAxisOrder(const std::string& order){
70 // North/South, East/West, Up/Downの3種類の軸を含んでいるかどうかを判定する。
71 if(order.size()!=3){
72 return false;
73 }
74 bool found[3] = {false,false,false}; // North/South, East/West, Up/Down
75 for(int i=0; i<3; ++i){
76 if(order.at(i)=='N'){ // North
77 found[0]=true;
78 }else if(order.at(i)=='S'){ // South
79 found[0]=true;
80 }else if(order.at(i)=='E'){ // East
81 found[1]=true;
82 }else if(order.at(i)=='W'){ // West
83 found[1]=true;
84 }else if(order.at(i)=='U'){ // UP
85 found[2]=true;
86 }else if(order.at(i)=='D'){ // DOWN
87 found[2]=true;
88 }
89 }
90 return found[0] && found[1] && found[2];

92Eigen::Matrix3d getAxisSwapRotationMatrix(const std::string& srcOrder_, const std::string& dstOrder_){
93 static const std::map<char,char> bodyCartesianReverseAxis={
94 {'F','A'},
95 {'A','F'},
96 {'S','P'},
97 {'P','S'},
98 {'U','D'},
99 {'D','U'}
100 };
101 static const std::map<char,char> topocentricReverseAxis={
102 {'N','S'},
103 {'S','N'},
104 {'E','W'},
105 {'W','E'},
106 {'U','D'},
107 {'D','U'}
108 };
109 std::string srcOrder=srcOrder_;
110 std::string dstOrder=dstOrder_;
111 if(checkBodyCartesianAxisOrder(srcOrder)){
112 if(!checkBodyCartesianAxisOrder(dstOrder)){
113 throw std::runtime_error("'"+srcOrder+"' and '"+dstOrder+"' are not same type of axisOrder.");
114 }
115 //RをSに、LをPに変換する。
116 for(int i=0; i<3; ++i){
117 if(srcOrder.at(i)=='R'){
118 srcOrder.at(i)='S';
119 }else if(srcOrder.at(i)=='L'){
120 srcOrder.at(i)='P';
121 }
122 if(dstOrder.at(i)=='R'){
123 dstOrder.at(i)='S';
124 }else if(dstOrder.at(i)=='L'){
125 dstOrder.at(i)='P';
126 }
127 }
128 Eigen::Matrix3d ret=Eigen::Matrix3d::Zero();
129 for(int i=0; i<3; ++i){
130 for(int j=0; j<3; ++j){
131 if(dstOrder.at(i)==srcOrder.at(j)){
132 ret(i,j)=1;
133 }else if(dstOrder.at(i)==bodyCartesianReverseAxis.at(srcOrder.at(j))){
134 ret(i,j)=-1;
135 }
136 }
137 }
138 return ret;
139 }else if(checkTopocentricCartesianAxisOrder(srcOrder)){
140 if(!checkTopocentricCartesianAxisOrder(srcOrder)){
141 throw std::runtime_error("'"+srcOrder+"' and '"+dstOrder+"' are not same type of axisOrder.");
142 }
143 Eigen::Matrix3d ret=Eigen::Matrix3d::Zero();
144 for(int i=0; i<3; ++i){
145 for(int j=0; j<3; ++j){
146 if(dstOrder.at(i)==srcOrder.at(j)){
147 ret(i,j)=1;
148 }else if(dstOrder.at(i)==topocentricReverseAxis.at(srcOrder.at(j))){
149 ret(i,j)=-1;
150 }
151 }
152 }
153 return ret;
154 }else{
155 throw std::runtime_error("Given axisOrder '"+srcOrder+"' is not valid as bodyCartesian or topocentric axisOrder.");
156 }
157}
158Quaternion getAxisSwapQuaternion(const std::string& srcOrder, const std::string& dstOrder){
159 return Quaternion::fromRotationMatrix(getAxisSwapRotationMatrix(std::move(srcOrder),std::move(dstOrder)));

357Quaternion CoordinateReferenceSystem::transformQuaternionTo(const Quaternion& q, const Eigen::Vector3d& location, const Time& time, const std::shared_ptr<CoordinateReferenceSystem>& dstCRS) const{
358 // 基準点locationにおいてこのCRSからの変換を表すクォータニオンをdstCRSからの変換を表すクォータニオンに変換する。
359 // q [another <- self]
360 // return [another <- dstCRS] = [another <- self] * [self <- dstCRS]
361 return q * getQuaternionTo(location,time,dstCRS).conjugate();
362}

365 // q [another <- srcCRS]
366 // return [another <- self] = [another <- srcCRS] * [srcCRS <- self]
367 return q * getQuaternionFrom(location,time,srcCRS).conjugate();

511Eigen::Vector3d CoordinateReferenceSystem::transformToTopocentricCRS(const Eigen::Vector3d& value, const Eigen::Vector3d& location, const Time& time, const CoordinateType& coordinateType, const Eigen::Vector3d& origin, const std::string& axisOrder, bool onSurface) const{
512 auto intermediateCRS = getNonDerivedBaseCRS()->getIntermediateCRS();
513 if(non_const_this() == intermediateCRS){
514 // 高度オフセットの計算(ENU経由)
515 auto qToSelfFromENU=getEquivalentQuaternionFromTopocentricCRS(origin,time,"ENU");
516 Eigen::Vector3d actualOriginInSelf;
517 if(onSurface){
518 auto altOffset=qToSelfFromENU.transformVector(
519 Eigen::Vector3d(0,0,getHeight(origin,time))
520 );// [Self <- ENU]
521 actualOriginInSelf=origin-fromCartesian(altOffset);
522 }else{
523 actualOriginInSelf=origin;
524 }
525 // クォータニオンの計算
526 Quaternion qToSelfFromTopo=getEquivalentQuaternionFromTopocentricCRS(origin,time,axisOrder);
527
528 // 変換の実行(MotionStateを利用)
529 MotionState topoMotionState(
530 non_const_this(),
531 time,
532 actualOriginInSelf,
533 Eigen::Vector3d::Zero(),
534 Eigen::Vector3d::Zero(),
535 qToSelfFromTopo,
536 "FSD"
537 );
538 if(coordinateType==CoordinateType::POSITION_ABS){
539 return topoMotionState.absPtoB(value);
540 }else if(coordinateType==CoordinateType::POSITION_REL){
541 return topoMotionState.relPtoB(value);
542 }else if(coordinateType==CoordinateType::DIRECTION){
543 return topoMotionState.dirPtoB(value);
544 }else if(coordinateType==CoordinateType::VELOCITY){
545 return topoMotionState.velPtoB(value,location);
546 }else if(coordinateType==CoordinateType::ANGULAR_VELOCITY){
547 return topoMotionState.omegaPtoB(value);
548 }else if(coordinateType==CoordinateType::FREE_VECTOR){
549 return topoMotionState.vecPtoB(value);
550 }else{
551 throw std::runtime_error("Invalide coordinate type.");
552 }
553 }else{
554 auto valueInIntermediate = transformTo(value,location,time,intermediateCRS,coordinateType);

579Eigen::Vector3d CoordinateReferenceSystem::transformFromTopocentricCRS(const Eigen::Vector3d& value, const Eigen::Vector3d& location, const Time& time, const CoordinateType& coordinateType, const Eigen::Vector3d& origin, const std::string& axisOrder, bool onSurface) const{
580 auto intermediateCRS = getNonDerivedBaseCRS()->getIntermediateCRS();
581 if(non_const_this() == intermediateCRS){
582 // 高度オフセットの計算(ENU経由)
583 auto qToSelfFromENU=getEquivalentQuaternionFromTopocentricCRS(origin,time,"ENU");
584 Eigen::Vector3d actualOriginInSelf;
585 if(onSurface){
586 auto altOffset=qToSelfFromENU.transformVector(
587 Eigen::Vector3d(0,0,getHeight(origin,time))
588 );// [Self <- ENU]
589 actualOriginInSelf=origin-fromCartesian(altOffset);
590 }else{
591 actualOriginInSelf=origin;
592 }
593 // クォータニオンの計算
594 Quaternion qToSelfFromTopo=getEquivalentQuaternionFromTopocentricCRS(origin,time,axisOrder);
595
596 // 変換の実行(MotionStateを利用)
597 MotionState topoMotionState(
598 non_const_this(),
599 time,
600 actualOriginInSelf,
601 Eigen::Vector3d::Zero(),
602 Eigen::Vector3d::Zero(),
603 qToSelfFromTopo,
604 "FSD"
605 );
606 if(coordinateType==CoordinateType::POSITION_ABS){
607 return topoMotionState.absBtoP(value);
608 }else if(coordinateType==CoordinateType::POSITION_REL){
609 return topoMotionState.relBtoP(value);
610 }else if(coordinateType==CoordinateType::DIRECTION){
611 return topoMotionState.dirBtoP(value);
612 }else if(coordinateType==CoordinateType::VELOCITY){
613 return topoMotionState.velBtoP(value,location);
614 }else if(coordinateType==CoordinateType::ANGULAR_VELOCITY){
615 return topoMotionState.omegaBtoP(value);
616 }else if(coordinateType==CoordinateType::FREE_VECTOR){
617 return topoMotionState.vecBtoP(value);
618 }else{
619 throw std::runtime_error("Invalide coordinate type.");
620 }
621 }else{
622 auto originInIntermediate = transformTo(origin,origin,time,intermediateCRS,CoordinateType::POSITION_ABS);
623 auto valueInIntermediate = intermediateCRS->transformFromTopocentricCRS(
624 value,
625 location,
626 time,
627 coordinateType,
628 originInIntermediate(),
629 axisOrder,
630 onSurface
631 );
632 auto locationInIntermediate = intermediateCRS->transformFromTopocentricCRS(
633 location,
634 location,
635 time,
636 CoordinateType::POSITION_ABS,
637 originInIntermediate(),
638 axisOrder,
639 onSurface
640 );
641 return transformFrom(valueInIntermediate,locationInIntermediate,time,intermediateCRS,coordinateType)();
642 }
