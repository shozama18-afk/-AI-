243void Track3D::merge(){
244 // このクラスでは算術平均でマージするが、必要に応じて派生クラスでオーバーライドすること。
245 if(buffer.size()>0){
246 Eigen::Vector3d meanPos=pos();
247 Eigen::Vector3d meanVel=vel();
248 for(auto& e:buffer){
249 Track3D extrapolated=e.transformTo(crs);
250 extrapolated.updateByExtrapolation(time-e.time);
251 meanPos+=extrapolated.pos();
252 meanVel+=extrapolated.vel();
253 }
254 meanPos/=(buffer.size()+1.0);
255 meanVel/=(buffer.size()+1.0);
256 _pos.setValue(meanPos);
257 _vel.setLocation(meanPos);
258 _vel.setValue(meanVel);
259 clearBuffer();
260 }

329Track2D::Track2D(const std::weak_ptr<PhysicalAsset>& truth_,const std::shared_ptr<CoordinateReferenceSystem>& crs_,const Eigen::Vector3d &origin_)
330:TrackBase(truth_,crs_),
331 _origin(origin_,crs,time,CoordinateType::POSITION_ABS),

333 _omega(Eigen::Vector3d::Zero(),origin_,crs,time,CoordinateType::FREE_VECTOR),
334 origin(_origin),dir(_dir),omega(_omega){
335 if(truth!=boost::uuids::nil_uuid()){
336 if(crs){
337 _origin=Coordinate(origin_,crs,time,CoordinateType::POSITION_ABS);

341 _omega=Coordinate(dir().cross(truth_.lock()->vel.asTypeOf(CoordinateType::FREE_VECTOR)(crs))/R,origin_,crs,time,CoordinateType::FREE_VECTOR);
342 }else{
343 _origin=Coordinate(origin_,crs,time,CoordinateType::POSITION_ABS);
344 Eigen::Vector3d dr=truth_.lock()->pos()-origin_;
345 double R=dr.norm();
346 _dir=Coordinate(dr/R,origin_,crs,time,CoordinateType::DIRECTION);
347 _omega=Coordinate(dir().cross(truth_.lock()->vel.asTypeOf(CoordinateType::FREE_VECTOR)())/R,origin_,crs,time,CoordinateType::FREE_VECTOR);
348 }

451double Track2D::getAZ() const{
452 // 方位角(真北を0、東向きを正)を返す
453 Eigen::Vector3d dirNED;
454 if(crs){
455 dirNED=crs->transformToTopocentricCRS(dir,origin,"NED",false);
456 }else{
457 dirNED=dir(); //親座標系が未指定の場合は暗黙にNEDを仮定する。
458 }
459 double xy=sqrt(dirNED(0)*dirNED(0)+dirNED(1)*dirNED(1));
460 if(xy==0){
461 return 0;
462 }else{
463 return atan2(dirNED(1),dirNED(0));
464 }
465}
466double Track2D::getEL() const{
467 // 仰角(水平を0、鉛直上向きを正)を返す
468 Eigen::Vector3d dirNED;
469 if(crs){
470 dirNED=crs->transformToTopocentricCRS(dir,origin,"NED",false);
471 }else{
472 dirNED=dir(); //親座標系が未指定の場合は暗黙にNEDを仮定する。
473 }
474 double xy=sqrt(dirNED(0)*dirNED(0)+dirNED(1)*dirNED(1));
475 if(xy==0){
476 return dirNED(2)>0 ? -M_PI_2 : M_PI_2;
477 }else{
478 return atan2(-dirNED(2),xy);
479 }

490void Track2D::merge(){
491 // このクラスでは算術平均でマージするが、必要に応じて派生クラスでオーバーライドすること。
492 // 例えば、観測点が複数ヶ所にわたる場合には明らかに不適切である。
493 if(buffer.size()>0){
494 Eigen::Vector3d meanOrigin=origin();
495 Eigen::Vector3d meanDir=dir();
496 Eigen::Vector3d meanOmega=omega();
497 for(auto& e:buffer){
498 Track2D extrapolated=e.transformTo(crs);
499 extrapolated.updateByExtrapolation(time-e.time);
500 meanDir+=extrapolated.dir();
501 meanOrigin+=extrapolated.origin();
502 meanOmega+=extrapolated.omega();
503 }
504 meanDir=(meanDir/(buffer.size()+1.0)).normalized();
505 meanOrigin/=(buffer.size()+1.0);
506 meanOmega/=(buffer.size()+1.0);
507 _origin.setValue(meanOrigin);
508 _dir.setLocation(meanOrigin);
509 _omega.setLocation(meanOrigin);
510 _dir.setValue(meanDir);
511 _omega.setValue(meanOmega);
512 clearBuffer();
513 }

524void Track2D::updateByExtrapolation(const double& dt){
525 // このクラスではcrs上の見かけの位置、角速度で外挿するが、必要に応じて派生クラスでオーバーライドすること。
526 // あるいは、慣性座標系等の外挿しても問題ないcrsに変換してから外挿すること
527 double w=omega.norm();
528 if(w>0){
529 Quaternion q=Quaternion::fromAngle(omega()/w,w*dt);
530 _dir.setValue(q.transformVector(dir()));
531 }
532 _time+=dt;

539 Quaternion q=Quaternion::fromAngle(omega()/w,w*dt);
540 return Track2D(truth,crs,time+dt,q.transformVector(dir()),origin(),omega());
541 }else{
